__device__ float wrapCoord(float x, int mode){
  if (mode==0 || mode==3) return fminf(fmaxf(x,0.f),1.f);       // clamp
  if (mode==2){ float t=floorf(x); float f=x-t; return ((int)t&1)?(1.f-f):f; } // mirror
  return x-floorf(x);                                            // repeat
}
__device__ float srgbToLinear(float x){
  return x<=0.04045f ? x*(1.f/12.92f) : powf((x+0.055f)*(1.f/1.055f),2.4f);
}
__device__ F4 sampleTexLevel(const unsigned char* texels, TexDesc td,
                             float u, float v){
  float su=wrapCoord(u,td.wrapS)*(td.width-1);
  float sv=wrapCoord(v,td.wrapT)*(td.height-1);
  int x0=(int)floorf(su), y0=(int)floorf(sv);
  int x1=(x0+1<td.width)?(x0+1):(td.width-1);
  int y1=(y0+1<td.height)?(y0+1):(td.height-1);
  float tx=su-x0, ty=sv-y0;
  const unsigned char* p00=texels+td.offset+(y0*td.width+x0)*4;
  const unsigned char* p10=texels+td.offset+(y0*td.width+x1)*4;
  const unsigned char* p01=texels+td.offset+(y1*td.width+x0)*4;
  const unsigned char* p11=texels+td.offset+(y1*td.width+x1)*4;
  F4 c00=mk4(p00[0]*(1.f/255.f),p00[1]*(1.f/255.f),p00[2]*(1.f/255.f),p00[3]*(1.f/255.f));
  F4 c10=mk4(p10[0]*(1.f/255.f),p10[1]*(1.f/255.f),p10[2]*(1.f/255.f),p10[3]*(1.f/255.f));
  F4 c01=mk4(p01[0]*(1.f/255.f),p01[1]*(1.f/255.f),p01[2]*(1.f/255.f),p01[3]*(1.f/255.f));
  F4 c11=mk4(p11[0]*(1.f/255.f),p11[1]*(1.f/255.f),p11[2]*(1.f/255.f),p11[3]*(1.f/255.f));
  if (td.srgb) {
    c00.x=srgbToLinear(c00.x); c00.y=srgbToLinear(c00.y); c00.z=srgbToLinear(c00.z);
    c10.x=srgbToLinear(c10.x); c10.y=srgbToLinear(c10.y); c10.z=srgbToLinear(c10.z);
    c01.x=srgbToLinear(c01.x); c01.y=srgbToLinear(c01.y); c01.z=srgbToLinear(c01.z);
    c11.x=srgbToLinear(c11.x); c11.y=srgbToLinear(c11.y); c11.z=srgbToLinear(c11.z);
  }
  F4 a=mk4(c00.x+(c10.x-c00.x)*tx,c00.y+(c10.y-c00.y)*tx,c00.z+(c10.z-c00.z)*tx,c00.w+(c10.w-c00.w)*tx);
  F4 b=mk4(c01.x+(c11.x-c01.x)*tx,c01.y+(c11.y-c01.y)*tx,c01.z+(c11.z-c01.z)*tx,c01.w+(c11.w-c01.w)*tx);
  return mk4(a.x+(b.x-a.x)*ty,a.y+(b.y-a.y)*ty,a.z+(b.z-a.z)*ty,a.w+(b.w-a.w)*ty);
}
__device__ F4 sampleTexRGBAFallbackLod(const unsigned char* texels,
                                      const TexDesc* tex, int numTex, int id,
                                      float u, float v, float lod, F4 missing){
  if (!texels || !tex || id<0 || id>=numTex) return missing;
  TexDesc td=tex[id];
  int baseId=id;
  if (td.isUdim) {
    int tileU=(int)floorf(u), tileV=(int)floorf(v);
    int lut=tileU+tileV*10;
    if (lut<0 || lut>=100) return missing;
    int tileId=td.udimLayer[lut];
    if (tileId<0 || tileId>=numTex) return missing;
    baseId=tileId; td=tex[tileId]; u=u-floorf(u); v=v-floorf(v);
  }
  if (td.width<=0 || td.height<=0) return missing;
  float cl=fminf(fmaxf(lod,0.f),(float)((td.mipCount>0?td.mipCount:1)-1));
  int l0=(int)floorf(cl), l1=(l0+1<td.mipCount)?l0+1:l0;
  int id0=l0==0?baseId:td.firstMip+l0-1;
  int id1=l1==0?baseId:td.firstMip+l1-1;
  TexDesc t0=l0==0?td:tex[id0];
  TexDesc t1=l1==0?td:tex[id1];
  F4 a=sampleTexLevel(texels,t0,u,v), b=sampleTexLevel(texels,t1,u,v);
  float f=cl-floorf(cl);
  return mk4(a.x+(b.x-a.x)*f,a.y+(b.y-a.y)*f,a.z+(b.z-a.z)*f,a.w+(b.w-a.w)*f);
}
__device__ F4 sampleTexRGBAFallback(const unsigned char* texels,
                                    const TexDesc* tex, int numTex, int id,
                                    float u, float v, F4 missing){
  return sampleTexRGBAFallbackLod(texels,tex,numTex,id,u,v,0.f,missing);
}
__device__ F4 sampleTexRGBA(const unsigned char* texels, const TexDesc* tex,
                            int numTex, int id, float u, float v){
  return sampleTexRGBAFallback(texels,tex,numTex,id,u,v,
                               mk4(1.f,1.f,1.f,1.f));
}

// Compact MaterialX graph interpreter shared by CUDA and HIP. Graph blocks
// are fixed-size, bounded, and dependency-first. Texture ids are already
// remapped to the resident table.
__device__ F4 evalMaterialXGraphUv(const float* graph, int mid, int route,
                                    float u, float v, float u1, float v1, int face,
                                 const unsigned char* texels,
                                 const TexDesc* textures, int numTextures,
                                 F3 normal=F3{0.f,0.f,1.f},
                                 F3 position=F3{0.f,0.f,0.f},
                                 F3 viewDirection=F3{0.f,0.f,1.f},
                                 float sceneTime=0.f,float sceneFrame=0.f,
                                 const GeomPropDesc* geomProps=0,
                                 const float* geomPropValues=0,
                                 int geomPropCount=0, int geomPropTri=-1) {
  const int stride = LIGHTUSD_GPU_GRAPH_FLOATS;
  const int base = mid * stride;
  if (!graph || mid < 0 || route < 0 || base + 7 >= 0x3fffffff) return mk4(0.f,0.f,0.f,0.f);
  int count = (int)fminf(fmaxf(graph[base], 0.f), (float)LIGHTUSD_GPU_GRAPH_MAX_NODES);
  int wanted = (int)floorf(graph[base + 1 + route] + 0.5f);
  if (count <= 0 || wanted < 0 || wanted >= count) return mk4(0.f,0.f,0.f,0.f);
  F4 values[LIGHTUSD_GPU_GRAPH_MAX_NODES];
  for (int i=0; i<count; ++i) {
      const int p=base+LIGHTUSD_GPU_GRAPH_HEADER_FLOATS+i*LIGHTUSD_GPU_GRAPH_NODE_FLOATS;
      const int op=(int)(graph[p]+0.5f);
      // floor(x + 0.5) preserves the packed -1 sentinel. A plain C cast
      // truncates -0.5 toward zero, turning constant inputs into self/slot-0
      // graph references and absent textures into texture zero.
      const int ia=(int)floorf(graph[p+1]+0.5f),
                ib=(int)floorf(graph[p+2]+0.5f),
                ic=(int)floorf(graph[p+3]+0.5f);
      const F4 a=(ia>=0&&ia<count)?values[ia]:mk4(graph[p+4],graph[p+5],graph[p+6],graph[p+7]);
      const F4 b=(ib>=0&&ib<count)?values[ib]:mk4(graph[p+8],graph[p+9],graph[p+10],graph[p+11]);
      const F4 c=(ic>=0&&ic<count)?values[ic]:mk4(graph[p+12],graph[p+13],graph[p+14],graph[p+15]);
      const F4 fallback=a;
      const int tex=(int)floorf(graph[p+16]+0.5f);
      if (op==0) values[i]=fallback;
      else if (op==123) {
        values[i]=fallback;
        if (geomProps && geomPropValues && geomPropTri >= 0) {
          unsigned int wanted=__float_as_uint(graph[p+17]);
          int components=(int)floorf(graph[p+18]+.5f);
          components=components<1?1:(components>16?16:components);
          int matrixColumn=(int)floorf(graph[p+19]+.5f);
          int outputComponents=components==9?3:(components==16?4:components);
          int matrixDim=components==9?3:4;
          matrixColumn=matrixColumn<0?0:(matrixColumn>=matrixDim?matrixDim-1:matrixColumn);
          float w0=1.f-u-v;
          for (int j=0; j<geomPropCount; ++j) {
            const GeomPropDesc d=geomProps[j];
            if (d.hash != wanted || (int)d.components != components) continue;
            const unsigned int baseValue=d.valueOffset +
                (unsigned int)geomPropTri * (unsigned int)(3 * components);
            unsigned int columnBase=(unsigned int)(matrixColumn*outputComponents);
            for (int lane=0; lane<outputComponents; ++lane)

                u * geomPropValues[baseValue + components + columnBase + lane] +

__device__ F4 samplePtexRGBA(const unsigned char* texels, const TexDesc* tex,
                             int numTex, int id, float u, float v,
                             unsigned int face, float lod){
  if (!tex || id<0 || id>=numTex || !tex[id].isPtex ||
      face>=(unsigned int)tex[id].ptexFaceCount)
    return sampleTexRGBALod(texels,tex,numTex,id,u,v,lod);
  const TexDesc td=tex[id];
  unsigned int val[4];
  #pragma unroll
  for (int c=0;c<4;++c) {
    int lo=td.ptexRectTexelOffset+(int)face*8+c*2;
    const unsigned char* a=texels+td.offset+lo*4;
    const unsigned char* b=a+4;
    val[c]=(unsigned int)a[3]|((unsigned int)b[3]<<8);
  }
  if (val[2]==0u || val[3]==0u) return mk4(1.f,1.f,1.f,1.f);
  float pu=fminf(fmaxf(u,0.f),1.f), pv=fminf(fmaxf(v,0.f),1.f);
  float px=(float)val[0]+pu*(float)(val[2]-1u);
  float py=(float)val[1]+(1.f-pv)*(float)(val[3]-1u);
  float au=td.width>1?px/(float)(td.width-1):0.f;
  float av=td.height>1?py/(float)(td.height-1):0.f;
  return sampleTexRGBALod(texels,tex,numTex,id,au,av,0.f);
}
__device__ F4 sampleMaterialTexture(const unsigned char* texels,
                                    const TexDesc* tex, int numTex, int id,
                                    float u, float v, unsigned int face,
                                    float lod, F4 missing) {
  if (tex && id >= 0 && id < numTex && tex[id].isPtex) {
    return samplePtexRGBA(texels, tex, numTex, id, u, v, face, lod);
  }
  return sampleTexRGBAFallbackLod(texels, tex, numTex, id, u, v, lod,
                                  missing);
}
__device__ float textureLod(const TexDesc* tex, int numTex, int id,
                            float u, float v, float uvFootprint){
  if (!tex || id<0 || id>=numTex) return 0.f;
  TexDesc td=tex[id];
  if (td.isUdim) {
    int lut=(int)floorf(u)+(int)floorf(v)*10;
    if (lut<0 || lut>=100 || td.udimLayer[lut]<0 ||
        td.udimLayer[lut]>=numTex) return 0.f;
    td=tex[td.udimLayer[lut]];
  }
  float rho=uvFootprint*(float)(td.width>td.height?td.width:td.height);
  float lod=log2f(fmaxf(rho,1.f));
  return fminf(fmaxf(lod,0.f),(float)((td.mipCount>0?td.mipCount:1)-1));
}
__device__ F4 sampleTexRGBALod(const unsigned char* texels, const TexDesc* tex,
                               int numTex, int id, float u, float v, float lod){
  return sampleTexRGBAFallbackLod(texels,tex,numTex,id,u,v,lod,
                                  mk4(1.f,1.f,1.f,1.f));
}
__device__ F3 sampleTexRGB(const unsigned char* texels, const TexDesc* tex,
                           int numTex, int id, float u, float v){
  F4 c=sampleTexRGBA(texels,tex,numTex,id,u,v);
  return mk(c.x,c.y,c.z);
}
__device__ F3 decodeRgbe(const unsigned char* p){
  if (!p || p[3]==0) return mk(0.f,0.f,0.f);
  float s=ldexpf(1.f,(int)p[3]-(128+8));
  return mk(((float)p[0]+.5f)*s,((float)p[1]+.5f)*s,
            ((float)p[2]+.5f)*s);
}
__device__ F3 sampleDomeEnv(const unsigned char* texels,const TexDesc* tex,
                            int numTex,int id,float u,float v,int rgbe){
  if (!rgbe) return sampleTexRGB(texels,tex,numTex,id,u,v);
  if (!texels||!tex||id<0||id>=numTex) return mk(0.f,0.f,0.f);
  TexDesc td=tex[id];
  if(td.width<=0||td.height<=0) return mk(0.f,0.f,0.f);
  u=wrapCoord(u,td.wrapS);v=wrapCoord(v,td.wrapT);
  float fx=u*(float)(td.width-1),fy=v*(float)(td.height-1);
  int x0=(int)floorf(fx),y0=(int)floorf(fy);
  int x1=x0+1<td.width?x0+1:x0,y1=y0+1<td.height?y0+1:y0;
  float tx=fx-floorf(fx),ty=fy-floorf(fy);
  F3 a=decodeRgbe(texels+td.offset+(y0*td.width+x0)*4);
  F3 b=decodeRgbe(texels+td.offset+(y0*td.width+x1)*4);
  F3 c=decodeRgbe(texels+td.offset+(y1*td.width+x0)*4);
  F3 e=decodeRgbe(texels+td.offset+(y1*td.width+x1)*4);
  F3 ab=add(a,scale(sub(b,a),tx)),ce=add(c,scale(sub(e,c),tx));
  return add(ab,scale(sub(ce,ab),ty));
}
__device__ float channel4(F4 c, int ch){
  if (ch==1) return c.y;
  if (ch==2) return c.z;
  if (ch==3) return c.w;
  return c.x;
}
__device__ void xformUv(const float* p, int off, float u, float v, float* ou, float* ov){
  const float* x=p+off;
  *ou=x[0]*u+x[1]*v+x[2];
  *ov=x[3]*u+x[4]*v+x[5];
}
__device__ float xformFootprint(const float* p, int off, float footprint){
  if (!p) return footprint;
  float sx=sqrtf(p[off+0]*p[off+0]+p[off+3]*p[off+3]);
  float sy=sqrtf(p[off+1]*p[off+1]+p[off+4]*p[off+4]);
  return footprint*fmaxf(sx,sy);
}

// Hash RNG + cosine-weighted hemisphere sampler for ambient occlusion.
__device__ unsigned hashU(unsigned x){ x^=x>>16; x*=0x7feb352du; x^=x>>15; x*=0x846ca68bu; x^=x>>16; return x; }
__device__ float rndf(unsigned& s){ s=s*747796405u+2891336453u; unsigned r=((s>>((s>>28)+4u))^s)*277803737u; return float((r>>22)^r)/4294967296.0f; }
// Uniformly sample an authored analytic light surface. A zero-sized shape is
// deliberately left at its representative position, preserving PointLight and
// legacy degenerate-area behavior exactly.
__device__ F3 sampleLightPosition(const float* lp, int lt, unsigned& seed){
  F3 c=mk(lp[4],lp[5],lp[6]);
  float radius=fmaxf(lp[7],0.f), width=fmaxf(lp[20],0.f);
  float height=fmaxf(lp[21],0.f), length=fmaxf(lp[22],0.f);
  F3 ax=norm3(mk(lp[40],lp[41],lp[42]));
  F3 ay=norm3(mk(lp[44],lp[45],lp[46]));
  F3 az=norm3(mk(lp[48],lp[49],lp[50]));
  if (lt==1 && radius>1e-7f) {
    float z=1.f-2.f*rndf(seed), a=6.28318530718f*rndf(seed);
    float r=sqrtf(fmaxf(0.f,1.f-z*z));
    return add(c,scale(mk(r*cosf(a),r*sinf(a),z),radius));
  }
  if (lt==2 && radius>1e-7f) {
    float a=6.28318530718f*rndf(seed), r=radius*sqrtf(rndf(seed));
    return add(c,add(scale(ax,r*cosf(a)),scale(ay,r*sinf(a))));
  }
  if (lt==3 && width>1e-7f && height>1e-7f) {
    return add(c,add(scale(ax,(rndf(seed)-.5f)*width),
                     scale(ay,(rndf(seed)-.5f)*height)));
  }
  if (lt==4 && radius>1e-7f && length>1e-7f) {
    float a=6.28318530718f*rndf(seed);
    return add(c,add(add(scale(ax,radius*cosf(a)),scale(ay,radius*sinf(a))),
                     scale(az,(rndf(seed)-.5f)*length)));
  }
  return c;
}
__device__ float iesFactor(const float* lp, F3 lightToPoint){
  if (!lp || fabsf(lp[52]-2.f)>0.25f) return 1.f;
  F3 axis=norm3(mk(lp[8],lp[9],lp[10]));
  float v=acosf(fminf(fmaxf(dot3(norm3(lightToPoint),axis),-1.f),1.f))*57.2957795f;
  float fy=fminf(fmaxf(v/60.f,0.f),3.f);
  int y=(int)floorf(fy), y1=y<3?y+1:3;
  float t=fy-(float)y;
  F3 d=norm3(lightToPoint);
  F3 axisX=norm3(mk(lp[40],lp[41],lp[42]));
  F3 axisY=norm3(mk(lp[44],lp[45],lp[46]));
  float az=atan2f(dot3(d,axisY),dot3(d,axisX))*57.2957795f;
  if (az<0.f) az+=360.f;
  float fx=az/60.f;
  int x0=((int)floorf(fx))%6, x1=(x0+1)%6;
  float tx=fx-floorf(fx);
  float a0=lp[56+y*6+x0]+(lp[56+y*6+x1]-lp[56+y*6+x0])*tx;
  float a1=lp[56+y1*6+x0]+(lp[56+y1*6+x1]-lp[56+y1*6+x0])*tx;
  return a0+(a1-a0)*t;
}
__device__ F3 domeShRadiance(const float* lights, int count, F3 dir){
  if (!lights) return mk(0.f,0.f,0.f);
  for (int i=0;i<count;++i) {
    const float* lp=&lights[i*80];
    if ((int)(lp[0]+0.5f)!=6) continue;
    if (!(((int)(lp[1]+0.5f))&(1<<7)))
      return mk(fmaxf(lp[16],0.f),fmaxf(lp[17],0.f),fmaxf(lp[18],0.f));
    F3 e=mk(lp[40]*dir.x+lp[41]*dir.y+lp[42]*dir.z,
            lp[44]*dir.x+lp[45]*dir.y+lp[46]*dir.z,
            lp[48]*dir.x+lp[49]*dir.y+lp[50]*dir.z);
    float b[9];
    b[0]=0.282095f; b[1]=0.488603f*e.y; b[2]=0.488603f*e.z;
    b[3]=0.488603f*e.x; b[4]=1.092548f*e.x*e.y;
    b[5]=1.092548f*e.y*e.z; b[6]=0.315392f*(3.f*e.z*e.z-1.f);
    b[7]=1.092548f*e.x*e.z; b[8]=0.546274f*(e.x*e.x-e.y*e.y);
    F3 out=mk(0.f,0.f,0.f);
    for (int k=0;k<9;++k) {
      out.x+=b[k]*lp[52+k*3+0]; out.y+=b[k]*lp[52+k*3+1];
      out.z+=b[k]*lp[52+k*3+2];
    }
    return mk(fmaxf(out.x,0.f)*lp[16],fmaxf(out.y,0.f)*lp[17],
              fmaxf(out.z,0.f)*lp[18]);
  }
  return mk(0.f,0.f,0.f);
}
__device__ F3 cosHemi(F3 n, float u1, float u2){
  float r=sqrtf(u1), phi=6.2831853f*u2;
  F3 t=norm3(fabsf(n.x)>0.9f?mk(0.f,1.f,0.f):mk(1.f,0.f,0.f));
  F3 b=norm3(cross3(n,t)); t=cross3(b,n);
  return norm3(add(add(scale(t,r*cosf(phi)),scale(b,r*sinf(phi))),scale(n,sqrtf(fmaxf(0.f,1.f-u1)))));
}

// column-major mat4 (invVP) times (ndc.x, ndc.y, z, 1) -> homogeneous xyz/w.
__device__ F3 unproject(const float* m, float nx, float ny, float z){
  float x = m[0]*nx + m[4]*ny + m[8]*z + m[12];
  float y = m[1]*nx + m[5]*ny + m[9]*z + m[13];
  float w3= m[2]*nx + m[6]*ny + m[10]*z + m[14];
  float w = m[3]*nx + m[7]*ny + m[11]*z + m[15];
  return mk(x/w, y/w, w3/w);
}

// Project a world point to pixel coords (column-major vp = proj*view). Returns
// (pixelX, pixelY, clipW); clipW<=0 means behind the eye. Pixel origin top-left,
// matching the primary-ray generation (ndcy -> (1-ndcy)/2 * H).
__device__ F3 projPix(const float* m, F3 p, float W, float H){
  float x = m[0]*p.x + m[4]*p.y + m[8]*p.z + m[12];
  float y = m[1]*p.x + m[5]*p.y + m[9]*p.z + m[13];
  float w = m[3]*p.x + m[7]*p.y + m[11]*p.z + m[15];
  float iw = (w>1e-9f||w<-1e-9f)?1.f/w:0.f;
  float sx = (x*iw*0.5f+0.5f)*W;
  float sy = (1.f-(y*iw*0.5f+0.5f))*H;
  return mk(sx, sy, w);
}
// 2D distance from point (px,py) to segment (a..b), in pixels.
__device__ float pixDistSeg(float px,float py,float ax,float ay,float bx,float by){
  float vx=bx-ax, vy=by-ay, wx=px-ax, wy=py-ay;
  float vv=vx*vx+vy*vy;
  float tt=vv>1e-9f?(wx*vx+wy*vy)/vv:0.f;
  tt=tt<0.f?0.f:(tt>1.f?1.f:tt);
  float dx=px-(ax+tt*vx), dy=py-(ay+tt*vy);
  return sqrtf(dx*dx+dy*dy);
}

// row-major 3x4 (o2w/w2o) * point / direction.
__device__ F3 xfPt(const float* m, F3 p){
  return mk(m[0]*p.x+m[1]*p.y+m[2]*p.z+m[3],
            m[4]*p.x+m[5]*p.y+m[6]*p.z+m[7],
            m[8]*p.x+m[9]*p.y+m[10]*p.z+m[11]);
}
__device__ F3 xfDir(const float* m, F3 v){
  return mk(m[0]*v.x+m[1]*v.y+m[2]*v.z,
            m[4]*v.x+m[5]*v.y+m[6]*v.z,
            m[8]*v.x+m[9]*v.y+m[10]*v.z);
}
// Object->world normal = (o2w^-1)^T * n = w2o^T * n (ignoring translation).
__device__ F3 xfNrm(const float* w2o, F3 n){
  return mk(w2o[0]*n.x+w2o[4]*n.y+w2o[8]*n.z,
            w2o[1]*n.x+w2o[5]*n.y+w2o[9]*n.z,
            w2o[2]*n.x+w2o[6]*n.y+w2o[10]*n.z);
}

__device__ bool sampleGeometryLight(const float* lp, const float* tris,
                                    const Inst* insts, const int* mats,
                                    const float* matPbr, int numMats,
                                    unsigned& seed, F3* outPos, F3* outN,
                                    F3* outEmission, float* outArea){
  if (!lp || !tris || !insts || !mats || !outPos || !outN || !outEmission || !outArea ||
      fabsf(lp[52]-3.f)>0.25f || lp[54] < 1.f || lp[55] < 0.f) return false;
  int triOffset=(int)(lp[53]+0.5f), triCount=(int)(lp[54]+0.5f);
  int instId=(int)(lp[55]+0.5f);
  if (triOffset<0 || triCount<=0 || instId<0) return false;
  const Inst* I=&insts[instId];
  int tri=-1;*outArea=0.f;
  for(int localTri=0;localTri<triCount;++localTri){
    int candidate=triOffset+localTri;const float* cv=&tris[candidate*9];
    F3 c0=xfPt(I->o2w,mk(cv[0],cv[1],cv[2]));
    F3 c1=xfPt(I->o2w,mk(cv[3],cv[4],cv[5]));
    F3 c2=xfPt(I->o2w,mk(cv[6],cv[7],cv[8]));
    float candidateArea=.5f*sqrtf(fmaxf(dot3(cross3(sub(c1,c0),sub(c2,c0)),cross3(sub(c1,c0),sub(c2,c0))),0.f));
    if(candidateArea<=1e-12f)continue;*outArea+=candidateArea;
    if(rndf(seed)<candidateArea/(*outArea))tri=candidate;
  }
  if(tri<0||*outArea<=1e-12f)return false;
  float su=sqrtf(rndf(seed)), bu=1.f-su, bv=rndf(seed)*su, bw=1.f-bu-bv;
  const float* tv=&tris[tri*9];
  F3 p0=mk(tv[0],tv[1],tv[2]), p1=mk(tv[3],tv[4],tv[5]), p2=mk(tv[6],tv[7],tv[8]);
  *outPos=xfPt(I->o2w,add(add(scale(p0,bu),scale(p1,bv)),scale(p2,bw)));
  F3 no=norm3(cross3(sub(p1,p0),sub(p2,p0)));
  *outN=norm3(xfNrm(I->w2o,no));
  int mid=mats[tri];
  if (mid>=0 && mid<numMats && matPbr) {
    *outEmission=mk(matPbr[mid*6+2],matPbr[mid*6+3],matPbr[mid*6+4]);
  } else {
    *outEmission=mk(lp[12],lp[13],lp[14]);
  }
  if (dot3(*outEmission,*outEmission)<1e-10f) *outEmission=mk(lp[12],lp[13],lp[14]);
  return true;
}

// slab test; returns true if [0,tmax] overlaps the node AABB along the ray.
__device__ bool hitAabb(const Node* nd, F3 o, F3 inv, float tmax){
  float t1=(nd->bmin[0]-o.x)*inv.x, t2=(nd->bmax[0]-o.x)*inv.x;
  float tmin=fminf(t1,t2), tmx=fmaxf(t1,t2);
  t1=(nd->bmin[1]-o.y)*inv.y; t2=(nd->bmax[1]-o.y)*inv.y;
  tmin=fmaxf(tmin,fminf(t1,t2)); tmx=fminf(tmx,fmaxf(t1,t2));
  t1=(nd->bmin[2]-o.z)*inv.z; t2=(nd->bmax[2]-o.z)*inv.z;
  tmin=fmaxf(tmin,fminf(t1,t2)); tmx=fminf(tmx,fmaxf(t1,t2));
  return tmx>=fmaxf(tmin,0.f) && tmin<tmax;
}

// Moller-Trumbore. Returns t (>0) on hit and the barycentrics, else -1.
__device__ float triHit(const float* tv, F3 o, F3 d, float* bu, float* bv){
  F3 p0=mk(tv[0],tv[1],tv[2]), p1=mk(tv[3],tv[4],tv[5]), p2=mk(tv[6],tv[7],tv[8]);
  F3 e1=sub(p1,p0), e2=sub(p2,p0);
  F3 pv=cross3(d,e2); float det=dot3(e1,pv);
  if (fabsf(det)<1e-12f) return -1.f;
  float inv=1.f/det;
  F3 tv0=sub(o,p0); float u=dot3(tv0,pv)*inv;
  if (u<0.f||u>1.f) return -1.f;
  F3 qv=cross3(tv0,e1); float v=dot3(d,qv)*inv;
  if (v<0.f||u+v>1.f) return -1.f;
  float t=dot3(e2,qv)*inv;
  if (t<=1e-4f) return -1.f;
  *bu=u; *bv=v; return t;
}

// Traverse one prototype's BLAS subtree (rooted at `root`) for the nearest hit in
// OBJECT space. leaf `left` is already a GLOBAL tri id (rebased at concat time).
__device__ float traverseBLAS(const Node* nodes, const float* tris, int root, F3 o,
                              F3 d, float tmax, int* hitTri, float* bu, float* bv,
                              bool anyHit){
  F3 inv = mk(1.f/d.x, 1.f/d.y, 1.f/d.z);
  int stack[64]; int sp=0; stack[sp++]=root;
  float best=tmax; *hitTri=-1;
  while (sp>0){
    const Node* nd=&nodes[stack[--sp]];
    if (!hitAabb(nd,o,inv,best)) continue;
    if (nd->count>0){
      for (int i=0;i<nd->count;i++){
        int ti=nd->left+i;
        float u,v; float t=triHit(&tris[ti*9],o,d,&u,&v);
        if (t>0.f && t<best){ best=t; *hitTri=ti; *bu=u; *bv=v; if (anyHit) return best; }
      }
    } else {
      stack[sp++]=nd->left;
      stack[sp++]=nd->right;
    }
  }
  return best;
}

// 2-level traversal: walk the TLAS; for each candidate instance transform the
// world ray into object space (no dir renorm, so object t == world t) and descend
// its BLAS. Tracks the nearest hit across instances + the hit instance index.
__device__ float traverseTLAS(const Node* tlas, const Inst* insts, const Node* blas,
                              const float* tris, F3 ow, F3 dw, float tmax,
                              int* hitTri, float* bu, float* bv, int* hitInst,
                              bool anyHit){
  F3 invW = mk(1.f/dw.x, 1.f/dw.y, 1.f/dw.z);
  int stack[48]; int sp=0; stack[sp++]=0;
  float best=tmax; *hitTri=-1; *hitInst=-1;
  while (sp>0){
    const Node* nd=&tlas[stack[--sp]];
    if (!hitAabb(nd,ow,invW,best)) continue;
    if (nd->count>0){
      for (int i=0;i<nd->count;i++){
        const Inst* I=&insts[nd->left+i];
        F3 oo=xfPt(I->w2o,ow), dd=xfDir(I->w2o,dw);  // un-normalized dd: t matches
        int lt; float lu,lv;
        float t=traverseBLAS(blas,tris,I->blasRoot,oo,dd,best,&lt,&lu,&lv,anyHit);
        if (lt>=0 && t<best){ best=t; *hitTri=lt; *bu=lu; *bv=lv; *hitInst=nd->left+i;
                              if (anyHit) return best; }
      }
    } else {
      stack[sp++]=nd->left;
      stack[sp++]=nd->right;
    }
  }
  return best;
}

// Count node visits (TLAS + visited BLAS subtrees) for the traversal-cost heatmap.
__device__ int traverseSteps(const Node* tlas, const Inst* insts, const Node* blas,
                             const float* tris, F3 ow, F3 dw, float tmax){
  F3 invW=mk(1.f/dw.x,1.f/dw.y,1.f/dw.z);
  int stack[48]; int sp=0; stack[sp++]=0; float best=tmax; int steps=0;
  while (sp>0){
    const Node* nd=&tlas[stack[--sp]]; steps++;
    if (!hitAabb(nd,ow,invW,best)) continue;
    if (nd->count>0){
      for (int i=0;i<nd->count;i++){
        const Inst* I=&insts[nd->left+i];
        F3 oo=xfPt(I->w2o,ow), dd=xfDir(I->w2o,dw);
        F3 invO=mk(1.f/dd.x,1.f/dd.y,1.f/dd.z);
        int bstack[64]; int bsp=0; bstack[bsp++]=I->blasRoot;
        while (bsp>0){
          const Node* bn=&blas[bstack[--bsp]]; steps++;
          if (!hitAabb(bn,oo,invO,best)) continue;
          if (bn->count>0){
            for (int j=0;j<bn->count;j++){ int ti=bn->left+j; float u,v; float t=triHit(&tris[ti*9],oo,dd,&u,&v); if (t>0.f&&t<best) best=t; }
          } else { bstack[bsp++]=bn->left; bstack[bsp++]=bn->right; }
        }
      }
    } else { stack[sp++]=nd->left; stack[sp++]=nd->right; }
  }
  return steps;
}

// --- UsdVol volume (dense float grid, object-space AABB) ---
// invModel: column-major mat4 (world->object). dim[3]=voxel dims, dim.w (packed
// as a 4th int) = float offset into the shared `volDens` buffer.
struct VolParam {
  float invModel[16];
  float bmin[4];      // .xyz aabb min, .w level-set selector
  float bmax[4];      // .xyz aabb max, .w isovalue
  int   dim[4];       // .xyz dims, .w = data offset (floats) into volDens
  float albedo[4];    // .xyz albedo, .w densityScale
  float emission[4];  // .xyz emission, .w background
};
__device__ F3 xfPt4(const float* m, F3 p){  // column-major mat4 * point
  float x = m[0]*p.x + m[4]*p.y + m[8]*p.z + m[12];
  float y = m[1]*p.x + m[5]*p.y + m[9]*p.z + m[13];
  float z = m[2]*p.x + m[6]*p.y + m[10]*p.z + m[14];
  float w = m[3]*p.x + m[7]*p.y + m[11]*p.z + m[15];
  if (w==0.f) w=1.f;
  return mk(x/w, y/w, z/w);
}
__device__ F3 xfDir4(const float* m, F3 v){
  return mk(m[0]*v.x + m[4]*v.y + m[8]*v.z,
            m[1]*v.x + m[5]*v.y + m[9]*v.z,
            m[2]*v.x + m[6]*v.y + m[10]*v.z);
}
__device__ float volFetch(const float* dens, int off, int dx, int dy, int dz,
                          int x, int y, int z){
  x = x<0?0:(x>=dx?dx-1:x);
  y = y<0?0:(y>=dy?dy-1:y);
  z = z<0?0:(z>=dz?dz-1:z);
  return dens[off + x + dx*(y + dy*z)];
}
__device__ float sampleVol(const float* dens, const VolParam* vp, F3 p){
  float ex=vp->bmax[0]-vp->bmin[0], ey=vp->bmax[1]-vp->bmin[1], ez=vp->bmax[2]-vp->bmin[2];
  if (ex<=0.f||ey<=0.f||ez<=0.f) return 0.f;
  int dx=vp->dim[0], dy=vp->dim[1], dz=vp->dim[2], off=vp->dim[3];
  float gx=(p.x-vp->bmin[0])/ex*dx-0.5f;
  float gy=(p.y-vp->bmin[1])/ey*dy-0.5f;
  float gz=(p.z-vp->bmin[2])/ez*dz-0.5f;
  int x0=(int)floorf(gx), y0=(int)floorf(gy), z0=(int)floorf(gz);
  float fx=gx-x0, fy=gy-y0, fz=gz-z0;
  float c00=volFetch(dens,off,dx,dy,dz,x0,y0,z0)*(1-fx)+volFetch(dens,off,dx,dy,dz,x0+1,y0,z0)*fx;
  float c10=volFetch(dens,off,dx,dy,dz,x0,y0+1,z0)*(1-fx)+volFetch(dens,off,dx,dy,dz,x0+1,y0+1,z0)*fx;
  float c01=volFetch(dens,off,dx,dy,dz,x0,y0,z0+1)*(1-fx)+volFetch(dens,off,dx,dy,dz,x0+1,y0,z0+1)*fx;
  float c11=volFetch(dens,off,dx,dy,dz,x0,y0+1,z0+1)*(1-fx)+volFetch(dens,off,dx,dy,dz,x0+1,y0+1,z0+1)*fx;
  float c0=c00*(1-fy)+c10*fy, c1=c01*(1-fy)+c11*fy;
  return c0*(1-fz)+c1*fz;
}
// Emission/absorption composite of one volume over background color `bg`.
__device__ F3 compositeVolume(const float* dens, const VolParam* vp, F3 worg, F3 wdir, F3 bg, float* nearest){
  F3 oo=xfPt4(vp->invModel, worg);
  F3 od=norm3(xfDir4(vp->invModel, wdir));
  F3 inv=mk(1.f/od.x, 1.f/od.y, 1.f/od.z);
  float t1a=(vp->bmin[0]-oo.x)*inv.x, t2a=(vp->bmax[0]-oo.x)*inv.x;
  float tmin=fminf(t1a,t2a), tmax=fmaxf(t1a,t2a);
  t1a=(vp->bmin[1]-oo.y)*inv.y; t2a=(vp->bmax[1]-oo.y)*inv.y;
  tmin=fmaxf(tmin,fminf(t1a,t2a)); tmax=fminf(tmax,fmaxf(t1a,t2a));
  t1a=(vp->bmin[2]-oo.z)*inv.z; t2a=(vp->bmax[2]-oo.z)*inv.z;
  tmin=fmaxf(tmin,fminf(t1a,t2a)); tmax=fminf(tmax,fmaxf(t1a,t2a));
  if (tmax<=fmaxf(tmin,0.f)) return bg;
)CUDA"
// NOTE: the kernel source is split into two adjacent raw-string literals here.
// MSVC caps a single string literal at 16380 bytes (C2026); the two pieces are
// concatenated by the compiler into one kKernelSrc. Keep the split on a line
// boundary so the NVRTC source is byte-identical.
R"CUDA(  float t0=fmaxf(tmin,0.f);
  float ex=vp->bmax[0]-vp->bmin[0], ey=vp->bmax[1]-vp->bmin[1], ez=vp->bmax[2]-vp->bmin[2];
  float vmin=fminf(ex/vp->dim[0], fminf(ey/vp->dim[1], ez/vp->dim[2]));
  float step=vmin*0.5f; if (step<=0.f) step=(tmax-t0)/256.f;
  if(vp->bmin[3]>0.5f){
    F3 objectDirection=xfDir4(vp->invModel,wdir);
    float speed=sqrtf(dot3(objectDirection,objectDirection));
    if(speed<=0.f||!isfinite(speed))return bg;
    tmax=fminf(tmax,(*nearest)*speed);
    if(tmax<=t0)return bg;
    float ds=fmaxf((tmax-t0)/512.f,vmin*.5f);
    float iso=vp->bmax[3],previous=sampleVol(dens,vp,add(oo,scale(od,t0)))-iso;
    float previousT=t0,hit=-1.f;
    for(int i=0;i<512;i++){
      float tn=fminf(t0+(i+1)*ds,tmax);
      float f=sampleVol(dens,vp,add(oo,scale(od,tn)))-iso;
      if((f<0.f&&previous>0.f)||(f>0.f&&previous<0.f)){
        float lo=previousT,hi=tn,flo=previous;
        for(int k=0;k<16;k++){
          float mid=.5f*(lo+hi);
          float fm=sampleVol(dens,vp,add(oo,scale(od,mid)))-iso;
          if(fm==0.f){lo=hi=mid;break;}
          if((fm>0.f)==(flo>0.f)){lo=mid;flo=fm;}else hi=mid;
        }
        hit=.5f*(lo+hi);break;
      }
      if(f!=0.f){previous=f;previousT=tn;}
      else if(previous==0.f)previousT=tn;
      if(tn>=tmax)break;
    }
    if(hit<0.f||hit/speed>=*nearest)return bg;
    F3 p=add(oo,scale(od,hit));
    float hx=ex/vp->dim[0],hy=ey/vp->dim[1],hz=ez/vp->dim[2];
    F3 g=mk((sampleVol(dens,vp,add(p,mk(hx,0,0)))-sampleVol(dens,vp,sub(p,mk(hx,0,0))))/hx,
            (sampleVol(dens,vp,add(p,mk(0,hy,0)))-sampleVol(dens,vp,sub(p,mk(0,hy,0))))/hy,
            (sampleVol(dens,vp,add(p,mk(0,0,hz)))-sampleVol(dens,vp,sub(p,mk(0,0,hz))))/hz);
    const float* m=vp->invModel;
    F3 n=norm3(mk(m[0]*g.x+m[1]*g.y+m[2]*g.z,
                  m[4]*g.x+m[5]*g.y+m[6]*g.z,
                  m[8]*g.x+m[9]*g.y+m[10]*g.z));
    F3 view=scale(norm3(wdir),-1.f);
    if(dot3(n,n)==0.f)n=view;
    float light=.15f+.85f*fabsf(dot3(n,view));
    *nearest=hit/speed;
    return add(scale(mk(vp->albedo[0],vp->albedo[1],vp->albedo[2]),light),
               mk(vp->emission[0],vp->emission[1],vp->emission[2]));
  }
  float scaleD=vp->albedo[3], background=vp->emission[3];
  float T=1.f; F3 L=mk(0.f,0.f,0.f);
  float t=t0;
  for (int i=0;i<256 && t<tmax;i++,t+=step){
    F3 p=add(oo, scale(od, t+0.5f*step));
    float dn=(sampleVol(dens,vp,p)-background)*scaleD;
    if (dn>0.f){
      float a=1.f-expf(-dn*step);
      L=add(L, scale(add(scale(mk(vp->albedo[0],vp->albedo[1],vp->albedo[2]),a),
                         scale(mk(vp->emission[0],vp->emission[1],vp->emission[2]),dn*step)), T));
      T*=(1.f-a);
      if (T<0.003f) break;
    }
  }
  return add(L, scale(bg, T));
}

// Alpha-mask intersections participate in traversal, not just compositing: a
// rejected texel must expose the next triangle and must not occlude AO/shadow
// rays. Blend materials remain intersections and composite after shading.
__device__ bool cutoutAllowsHit(
    int ht, float bu, float bv, int hitInst, const float* cols,
    const float* tris, F3 rayDir, const int* mats, const int* backMats,
    const float* matPbr, const float* matLightRt, const float* matGraph,
    int numMats,
    const int* matTex, const float* matTexParam, const unsigned char* texels,
    const TexDesc* textures, int numTextures, const float* uvs,
    const float* uvs1, const Inst* insts) {
  int mid=mats[ht];
  if (backMats && backMats[ht]>=0) {
    const float* tv=&tris[ht*9];
    const Inst* I=&insts[hitInst];
    F3 p0=xfPt(I->o2w,mk(tv[0],tv[1],tv[2]));
    F3 p1=xfPt(I->o2w,mk(tv[3],tv[4],tv[5]));
    F3 p2=xfPt(I->o2w,mk(tv[6],tv[7],tv[8]));
    if (dot3(cross3(sub(p1,p0),sub(p2,p0)),rayDir)>0.f) mid=backMats[ht];
  }
  if (mid<0 || mid>=numMats || !matLightRt) return true;
  const float* lrt=&matLightRt[mid*LIGHTUSD_GPU_OPENPBR_FLOATS];
  float alphaMode=lrt[54];
  if (!(alphaMode>0.5f && alphaMode<1.5f)) return true;
  float w0=1.f-bu-bv;
  const float* uvh=&uvs[ht*6];
  float u=uvh[0]*w0+uvh[2]*bu+uvh[4]*bv;
  float v=uvh[1]*w0+uvh[3]*bu+uvh[5]*bv;
  float u1=u,v1=v;
  if (uvs1){
    const float* uvh1=&uvs1[ht*6];
    u1=uvh1[0]*w0+uvh1[2]*bu+uvh1[4]*bv;
    v1=uvh1[1]*w0+uvh1[3]*bu+uvh1[5]*bv;
  }
  const float* tp=matTexParam?&matTexParam[mid*LIGHTUSD_GPU_MATERIAL_TEX_PARAM_FLOATS]:0;
  int uvbits=tp?(int)(tp[69]+0.5f):0;
  float buu=u,bvv=v,ouu=u,ovv=v;
  if (tp){
    xformUv(tp,0,(uvbits&1)?u1:u,(uvbits&1)?v1:v,&buu,&bvv);
    xformUv(tp,30,(uvbits&32)?u1:u,(uvbits&32)?v1:v,&ouu,&ovv);
  }
  int baseTex=matTex?matTex[mid*LIGHTUSD_GPU_MATERIAL_TEX_SLOTS+0]:-1;
  F4 bt=sampleMaterialTexture(texels,textures,numTextures,baseTex,buu,bvv,
                              (unsigned int)ht,0.f,mk4(1.f,1.f,1.f,1.f));
  if (tp) bt=mk4(bt.x*tp[36]+tp[40],bt.y*tp[37]+tp[41],
                 bt.z*tp[38]+tp[42],bt.w*tp[39]+tp[43]);
  float opacity=(lrt[51]>0.5f)?lrt[39]:matPbr[mid*6+5];
  opacity*=bt.w;
  int opacityTex=matTex?matTex[mid*LIGHTUSD_GPU_MATERIAL_TEX_SLOTS+5]:-1;
  if (opacityTex>=0){
    F4 ot=sampleMaterialTexture(texels,textures,numTextures,opacityTex,ouu,ovv,
                                (unsigned int)ht,0.f,mk4(1.f,1.f,1.f,1.f));
    int ch=tp?(int)(tp[66]+0.5f):0;
    opacity*=channel4(ot,ch)*(tp?tp[67]:1.f)+(tp?tp[68]:0.f);
  }
  if (matGraph && matGraph[mid*LIGHTUSD_GPU_GRAPH_FLOATS+4] >= 0.f) {
    opacity=evalMaterialXGraph(matGraph,mid,3,ouu,ovv,(unsigned int)ht,
                               texels,textures,numTextures).x;
  }
  const float* cv=&cols[ht*12];
  opacity*=(cv[3]*w0+cv[7]*bu+cv[11]*bv)*insts[hitInst].tint[3];
  return opacity>=lrt[55];
}

__device__ float traverseCutout(
    const Node* tlas, const Inst* insts, const Node* blas, const float* tris,
    F3 o, F3 d, float tmax, int* ht, float* bu, float* bv, int* hitInst,
    const float* cols, const int* mats, const int* backMats, const float* matPbr,
    const float* matLightRt, const float* matGraph, int numMats,
    const int* matTex,
    const float* matTexParam, const unsigned char* texels,
    const TexDesc* textures, int numTextures, const float* uvs,
    const float* uvs1, int shadowLight, bool shadowLinksAll) {
  float base=0.f;
  for (int pass=0; pass<64 && base<tmax; ++pass){
    F3 ro=add(o,scale(d,base));
    float t=traverseTLAS(tlas,insts,blas,tris,ro,d,tmax-base,
                         ht,bu,bv,hitInst,false);
    if (*ht<0) return tmax;
    if (shadowLight >= 0) {
      bool linked = shadowLight < 32
          ? ((insts[*hitInst].shadowLightMask & (1u << shadowLight)) != 0)
          : shadowLinksAll;
      if (!linked) { base += t + 1e-4f; continue; }
    }
    if (cutoutAllowsHit(*ht,*bu,*bv,*hitInst,cols,tris,d,mats,backMats,
                        matPbr,matLightRt,matGraph,
                        numMats,matTex,matTexParam,texels,textures,numTextures,
                        uvs,uvs1,insts)) return base+t;
    base+=t+1e-4f;
  }
  *ht=-1; *hitInst=-1;
  return tmax;
}

__device__ F3 ptCosine(F3 n,unsigned& seed){
  float r=sqrtf(rndf(seed)),p=6.28318530718f*rndf(seed);
  F3 a=fabsf(n.z)<.999f?mk(0.f,0.f,1.f):mk(0.f,1.f,0.f);
  F3 t=norm3(cross3(a,n)),b=cross3(n,t);
  return norm3(add(add(scale(t,r*cosf(p)),scale(b,r*sinf(p))),
                   scale(n,sqrtf(fmaxf(0.f,1.f-r*r)))));
}

__device__ void ptBasis(F3 n,F3* t,F3* b){
  F3 a=fabsf(n.z)<.999f?mk(0.f,0.f,1.f):mk(0.f,1.f,0.f);
  *t=norm3(cross3(a,n));*b=cross3(n,*t);
}
__device__ void ptRotatedBasis(F3 n,float rotation,F3* t,F3* b){F3 t0,b0;ptBasis(n,&t0,&b0);float a=rotation*.0174532925199433f;*t=add(scale(t0,cosf(a)),scale(b0,sinf(a)));*b=add(scale(t0,-sinf(a)),scale(b0,cosf(a)));}
// Sample the same isotropic GGX NDF used by ggxD().  Returning the matching
// reflected-direction PDF is essential: blending mirror and cosine directions
// (the old implementation) has no tractable PDF and biased every glossy path.
__device__ F3 ptSampleGgx(F3 n,F3 v,float rough,float anisotropy,float rotation,unsigned& seed,float* pdf){
  float alpha=fmaxf(rough*rough,.002f),ax=fmaxf(alpha*(1.f+anisotropy),.001f),ay=fmaxf(alpha*(1.f-anisotropy),.001f);
  float u1=rndf(seed),u2=rndf(seed);
  float phi=atanf(ay/ax*tanf(6.28318530718f*u2));if(u2>.25f&&u2<=.75f)phi+=3.14159265359f;else if(u2>.75f)phi+=6.28318530718f;float cp=cosf(phi),sp=sinf(phi);float a2=1.f/(cp*cp/(ax*ax)+sp*sp/(ay*ay));
  float cosTheta=sqrtf(fmaxf((1.f-u1)/(1.f+(a2-1.f)*u1),0.f));
  float sinTheta=sqrtf(fmaxf(1.f-cosTheta*cosTheta,0.f));
  F3 t,b;ptRotatedBasis(n,rotation,&t,&b);
  F3 h=norm3(add(add(scale(t,sinTheta*cp),scale(b,sinTheta*sp)),scale(n,cosTheta)));
  if(dot3(h,v)<0.f)h=scale(h,-1.f);
  F3 l=ptReflect(scale(v,-1.f),h);
  float nh=fmaxf(dot3(n,h),0.f),vh=fmaxf(dot3(v,h),1e-6f);
  *pdf=ggxDAniso(nh,dot3(h,t),dot3(h,b),rough,anisotropy)*nh/fmaxf(4.f*vh,1e-6f);
  return norm3(l);
}
__device__ float ptGgxDirectionPdf(F3 n,F3 v,F3 l,float rough,float anisotropy,float rotation){
  if(dot3(n,l)<=0.f)return 0.f;F3 h=norm3(add(v,l));
  float nh=fmaxf(dot3(n,h),0.f),vh=fmaxf(dot3(v,h),1e-6f);
  F3 t,b;ptRotatedBasis(n,rotation,&t,&b);return ggxDAniso(nh,dot3(h,t),dot3(h,b),rough,anisotropy)*nh/fmaxf(4.f*vh,1e-6f);
}
__device__ float ptPowerHeuristic(float a,float b){
  float aa=a*a,bb=b*b;return aa/fmaxf(aa+bb,1e-20f);
}
__device__ F3 ptEvaluateReflection(PtMat s,F3 n,F3 coatN,F3 v,F3 l,
    F3 diffuseC,F3 f0,F3 sheenC,F3 coatF0,float coatWeight,
    float diffuseP,float glossyP,float coatP,
    float* pdf){
  float nl=fmaxf(dot3(n,l),0.f),nv=fmaxf(dot3(n,v),1e-6f);
  if(nl<=0.f){*pdf=0.f;return mk(0.f,0.f,0.f);}F3 h=norm3(add(v,l));
  float nh=fmaxf(dot3(n,h),0.f),vh=fmaxf(dot3(v,h),0.f);
  F3 F=fresnelSchlick(vh,f0);F3 t,b;ptRotatedBasis(n,s.anisotropyRotation,&t,&b);
  float gs=ggxDAniso(nh,dot3(h,t),dot3(h,b),s.rough,s.anisotropy)*ggxG1Aniso(nv,dot3(v,t),dot3(v,b),s.rough,s.anisotropy)*ggxG1Aniso(nl,dot3(l,t),dot3(l,b),s.rough,s.anisotropy)/fmaxf(4.f*nv*nl,1e-6f);
  F3 ch=norm3(add(v,l)),ct,cb;ptRotatedBasis(coatN,s.coatRotation,&ct,&cb);float cnl=fmaxf(dot3(coatN,l),0.f),cnv=fmaxf(dot3(coatN,v),1e-6f),cnh=fmaxf(dot3(coatN,ch),0.f);F3 cF=scale(fresnelSchlick(fmaxf(dot3(v,ch),0.f),coatF0),coatWeight);
  float cs=ggxDAniso(cnh,dot3(ch,ct),dot3(ch,cb),s.coatRoughness,s.coatAnisotropy)*ggxG1Aniso(cnv,dot3(v,ct),dot3(v,cb),s.coatRoughness,s.coatAnisotropy)*ggxG1Aniso(fmaxf(cnl,1e-6f),dot3(l,ct),dot3(l,cb),s.coatRoughness,s.coatAnisotropy)/fmaxf(4.f*cnv*fmaxf(cnl,1e-6f),1e-6f);
  *pdf=diffuseP*nl*.31830988618f+
       glossyP*ptGgxDirectionPdf(n,v,l,s.rough,s.anisotropy,s.anisotropyRotation)+
       coatP*ptGgxDirectionPdf(coatN,v,l,s.coatRoughness,s.coatAnisotropy,s.coatRotation);
  float sheenShape=(.5f+.5f*s.sheenRoughness)*powf(1.f-nl,5.f)*.31830988618f;
  F3 diffuse=mk((1.f-F.x)*diffuseC.x*.31830988618f,(1.f-F.y)*diffuseC.y*.31830988618f,(1.f-F.z)*diffuseC.z*.31830988618f);return add(add(diffuse,scale(sheenC,sheenShape)),
             add(scale(F,gs),scale(cF,cs)));
}
__device__ F3 ptSampleHg(F3 forward,float g,unsigned& seed){
  g=fminf(fmaxf(g,-.95f),.95f);float u=rndf(seed),cosTheta;
  if(fabsf(g)<1e-3f)cosTheta=1.f-2.f*u;
  else{float q=(1.f-g*g)/(1.f-g+2.f*g*u);cosTheta=(1.f+g*g-q*q)/(2.f*g);}
  cosTheta=fminf(fmaxf(cosTheta,-1.f),1.f);
  float phi=6.28318530718f*rndf(seed),sinTheta=sqrtf(fmaxf(0.f,1.f-cosTheta*cosTheta));
  F3 t,b;ptBasis(norm3(forward),&t,&b);
  return norm3(add(add(scale(t,sinTheta*cosf(phi)),scale(b,sinTheta*sinf(phi))),
                   scale(norm3(forward),cosTheta)));
}
__device__ float ptObservedToScatterChannel(float observed,float g){
  float c=fminf(fmaxf(observed,0.f),.999999f);
  float s=4.09712f+4.20863f*c-sqrtf(9.59217f+41.6808f*c+17.7126f*c*c);
  float s2=s*s;
  return fminf(fmaxf((1.f-s2)/fmaxf(1.f-g*s2,1e-6f),0.f),.999999f);
}
__device__ F3 ptObservedToScatterAlbedo(F3 observed,float g){

            ptObservedToScatterChannel(observed.y,g),
            ptObservedToScatterChannel(observed.z,g));
}

__device__ F3 ptEnvironment(const float* lights,int count,F3 d,F3 clear,
                            const unsigned char* texels,const TexDesc* textures,
                            int numTextures){
  if(lights&&texels&&textures){for(int i=0;i<count;++i){const float* lp=&lights[i*80];int flags=(int)(lp[1]+.5f);if((int)(lp[0]+.5f)!=6||!(flags&(1<<4)))continue;F3 q=norm3(mk(lp[40]*d.x+lp[41]*d.y+lp[42]*d.z,lp[44]*d.x+lp[45]*d.y+lp[46]*d.z,lp[48]*d.x+lp[49]*d.y+lp[50]*d.z));float u=atan2f(q.x,-q.z)*.15915494f+.5f,v=acosf(fminf(fmaxf(q.y,-1.f),1.f))*.31830989f;F3 c=sampleDomeEnv(texels,textures,numTextures,(int)(lp[2]+.5f),u,v,(flags&(1<<8))!=0);return mk(c.x*lp[16],c.y*lp[17],c.z*lp[18]);}}
  F3 e=domeShRadiance(lights,count,d);
  // The viewer clear color is display-space, while path throughput and the
  // accumulation buffer are scene-linear. Match the Vulkan path's conversion
  // so CUDA/HIP EXR output and background radiance have identical semantics.
  return dot3(e,e)>1e-12f?e:mk(srgbToLinear(clear.x),srgbToLinear(clear.y),
                               srgbToLinear(clear.z));
}
__device__ F3 ptEnvironmentCellDirection(int cell,float jx,float jy){
  int zCell=cell/8,phiCell=cell-zCell*8;
  float z=-1.f+2.f*((float)zCell+jy)/4.f;
  float phi=6.28318530718f*((float)phiCell+jx)/8.f;
  float r=sqrtf(fmaxf(0.f,1.f-z*z));return mk(r*cosf(phi),r*sinf(phi),z);
}
__device__ float ptEnvironmentCellWeight(const float* lights,int count,
                                          F3 clear,int cell,const unsigned char* texels,
                                          const TexDesc* textures,int numTextures){
  // The host caches total/max values from these same HDR texels, so rejection
  // sampling and MIS use one exact discrete proposal instead of an SH proxy.
  F3 e=ptEnvironment(lights,count,
      ptEnvironmentCellDirection(cell,.5f,.5f),clear,texels,textures,numTextures);
  return fmaxf(ptLum(e),1e-6f);
}
__device__ float ptEnvironmentWeightTotal(const float* lights,int count,F3 clear,
                                           const unsigned char* texels,
                                           const TexDesc* textures,int numTextures){
  if(lights){for(int i=0;i<count;++i){const float* lp=&lights[i*80];
    if((int)(lp[0]+.5f)==6&&lp[79]>0.f)return lp[79];}}
  float total=0.f;
  for(int cell=0;cell<32;++cell)
    total+=ptEnvironmentCellWeight(lights,count,clear,cell,texels,textures,numTextures);
  return fmaxf(total,1e-20f);
}
__device__ float ptEnvironmentWeightMax(const float* lights,int count){
  if(lights){for(int i=0;i<count;++i){const float* lp=&lights[i*80];
    if((int)(lp[0]+.5f)==6&&lp[51]>0.f)return lp[51];}}
  return 0.f;
}
__device__ F3 ptSampleEnvironment(const float* lights,int count,F3 clear,
                                  unsigned& seed,float* pdf,const unsigned char* texels,
                                  const TexDesc* textures,int numTextures,
                                  float total,float maxWeight){
  int selected=0;float selectedWeight=1e-6f;
  if(maxWeight>0.f){
    // Exact discrete rejection sampling of the same 32-cell distribution as
    // the former reservoir scan. The CPU-packed SH maximum makes the average
    // cost max/mean cell evaluations instead of an unconditional 32.
    do{selected=min((int)(rndf(seed)*32.f),31);selectedWeight=ptEnvironmentCellWeight(lights,count,clear,selected,texels,textures,numTextures);}while(rndf(seed)*maxWeight>selectedWeight);
  }else{
    float running=0.f;
    for(int cell=0;cell<32;++cell){float w=ptEnvironmentCellWeight(lights,count,clear,cell,texels,textures,numTextures);running+=w;if(rndf(seed)<w/running){selected=cell;selectedWeight=w;}}
  }
  *pdf=selectedWeight/total*2.5464790894703255f;
  return ptEnvironmentCellDirection(selected,rndf(seed),rndf(seed));
}
__device__ float ptEnvironmentPdf(const float* lights,int count,F3 clear,
                                  F3 direction,const unsigned char* texels,
                                  const TexDesc* textures,int numTextures,
                                  float total){
  direction=norm3(direction);float phi=atan2f(direction.y,direction.x);
  if(phi<0.f)phi+=6.28318530718f;
  int px=(int)(phi*(8.f/6.28318530718f));if(px>7)px=7;
  int pz=(int)((direction.z+1.f)*2.f);if(pz<0)pz=0;else if(pz>3)pz=3;
  int selected=pz*8+px;
  return ptEnvironmentCellWeight(lights,count,clear,selected,texels,textures,numTextures)/fmaxf(total,1e-20f)*2.5464790894703255f;
}
__device__ float ptSpectralIor(float iorD,float abbe,float dispersion,int channel){if(dispersion<=0.f||abbe<=0.f)return iorD;float lc=.6562725f,ld=.5875618f,lf=.4861327f,delta=(iorD-1.f)/fmaxf(abbe,1.f)*dispersion;float ic=1.f/(lc*lc),id=1.f/(ld*ld),iff=1.f/(lf*lf),B=delta/fmaxf(iff-ic,1e-6f),A=iorD-B*id,lambda=channel==0?lc:(channel==1?ld:lf);return fmaxf(A+B/(lambda*lambda),1.0001f);}

      float prob=ws/sum;
      w=scale(subsurfaceC,1.f/fmaxf(prob,1e-6f));
      subsurfaceG=fminf(fmaxf(m.subsurfaceAnisotropy+0.5f*m.subsurfaceScatterAnisotropy,-.95f),.95f);subsurfaceAlbedo=ptObservedToScatterAlbedo(m.subsurfaceColor,subsurfaceG);float radius=fmaxf(m.radius.x,fmaxf(m.radius.y,m.radius.z));subsurfaceSigma=1.f/fmaxf(radius,1e-5f);
      ro=add(p,scale(n,-1e-4f));rd=ptCosine(scale(n,-1.f),seed);
      beta=ptMul(beta,w);inSubsurface=true;previousDelta=true;++subsurfaceEvents;continue;
    }else if((xi-=wc)<0.f){
      float pdf=0.f;F3 v=scale(rd,-1.f);
      nd=ptSampleGgx(coatN,v,m.coatRoughness,m.coatAnisotropy,m.coatRotation,seed,&pdf);
      float nl=fmaxf(dot3(coatN,nd),0.f);if(nl<=0.f||pdf<=1e-8f)break;reflectionSample=true;previousDelta=false;
    }else if((xi-=wg)<0.f){
      float pdf=0.f;F3 v=scale(rd,-1.f);
      nd=ptSampleGgx(n,v,m.rough,m.anisotropy,m.anisotropyRotation,seed,&pdf);
      float nl=fmaxf(dot3(n,nd),0.f);if(nl<=0.f||pdf<=1e-8f)break;reflectionSample=true;previousDelta=false;
    }else{
      nd=ptCosine(n,seed);reflectionSample=true;previousDelta=false;
    }
    if(reflectionSample){float mixturePdf=0.f;F3 reflection=ptEvaluateReflection(m,n,coatN,scale(rd,-1.f),nd,
        diffuseC,f0,sheenC,coatF0,m.coatWeight,diffuseP,glossyP,coatP,&mixturePdf);float nl=fmaxf(dot3(n,nd),0.f);if(mixturePdf<=1e-8f||nl<=0.f)break;w=scale(reflection,nl/mixturePdf);previousBsdfPdf=mixturePdf;previousEnvironmentPdf=ptEnvironmentPdf(lightParams,numLights,clear,nd,texels,textures,numTextures,environmentWeightTotal);
    }else{previousBsdfPdf=0.f;previousEnvironmentPdf=0.f;}
    beta=ptMul(beta,w);if(ptMax(beta)<1e-5f)break;
    bool transmissionRoulette=false;
    // Interactive paths commonly finish their useful dielectric work after
    // entering and leaving one shell. Continue deeper paths stochastically
    // (with reciprocal weighting, so the estimator remains unbiased) instead
    // of carrying almost every glass ray through bounces five and six. Final
    // quality uses maxDepth>6 and deliberately keeps the full path budget.
    if(maxDepth<=6&&transmissionEvents>=2&&bounce+1>=3){
      const float survive=.625f;
      if(rndf(seed)>survive)break;
      beta=scale(beta,1.f/survive);transmissionRoulette=true;
    }
    if(!transmissionRoulette&&bounce+1>=rrDepth){float survive=fminf(fmaxf(ptMax(beta),.05f),.95f);if(rndf(seed)>survive)break;beta=scale(beta,1.f/survive);}if(side!=0.f)ro=add(p,scale(geomN,side*1e-4f));rd=nd;
  }
  return L;
}

// Bounded underlay shading for primary alpha layers. The nearest layer keeps
// the full OpenPBR preview above; deeper layers use resolved albedo/emission
// so stacked blend materials retain correct coverage without duplicating the
// complete direct-lighting path for every hit.
__device__ F3 approxPrimaryLayer(
    int ht, float bu, float bv, int hitInst, const float* cols,
    const int* mats, const float* matBase, const float* matPbr,
    const float* matLightRt, const float* matGraph, int numMats, const int* matTex,
    const float* matTexParam, const unsigned char* texels,
    const TexDesc* textures, int numTextures, const float* uvs,
    const Inst* insts) {
  int mid = mats[ht];
  F3 base = mk(0.6f, 0.6f, 0.6f);
  F3 emission = mk(0.f, 0.f, 0.f);
  if (mid >= 0 && mid < numMats) {
    if (matBase) base = mk(matBase[mid * 3], matBase[mid * 3 + 1],
                            matBase[mid * 3 + 2]);
    if (matLightRt && matLightRt[mid * LIGHTUSD_GPU_OPENPBR_FLOATS + 51] > 0.5f)
      base = scale(mk(matLightRt[mid * LIGHTUSD_GPU_OPENPBR_FLOATS], matLightRt[mid * LIGHTUSD_GPU_OPENPBR_FLOATS + 1],
                      matLightRt[mid * LIGHTUSD_GPU_OPENPBR_FLOATS + 2]), matLightRt[mid * LIGHTUSD_GPU_OPENPBR_FLOATS + 3]);
    if (matPbr) emission = mk(matPbr[mid * 6 + 2], matPbr[mid * 6 + 3],
                              matPbr[mid * 6 + 4]);
    const float* uv = &uvs[ht * 6];
    float u = uv[0] * (1.f - bu - bv) + uv[2] * bu + uv[4] * bv;
    float v = uv[1] * (1.f - bu - bv) + uv[3] * bu + uv[5] * bv;
    int tex = matTex ? matTex[mid * LIGHTUSD_GPU_MATERIAL_TEX_SLOTS] : -1;
    F4 t = sampleMaterialTexture(texels, textures, numTextures, tex, u, v,
                                 (unsigned int)ht, 0.f,
                                 mk4(1.f, 1.f, 1.f, 1.f));
    base = mk(base.x * t.x, base.y * t.y, base.z * t.z);
    if (matGraph && matGraph[mid * LIGHTUSD_GPU_GRAPH_FLOATS + 1] >= 0.f) {
      F4 g = evalMaterialXGraph(matGraph, mid, 0, u, v, ht, texels,
                                textures, numTextures);
      base = mk(g.x, g.y, g.z);
    }
    if (matTex && matTexParam) {
      int et = matTex[mid * LIGHTUSD_GPU_MATERIAL_TEX_SLOTS + 4];
      const float* p = &matTexParam[mid * LIGHTUSD_GPU_MATERIAL_TEX_PARAM_FLOATS];
      F4 e = sampleMaterialTexture(texels, textures, numTextures, et, u, v,
                                   (unsigned int)ht, 0.f,
                                   mk4(1.f, 1.f, 1.f, 1.f));
      e = mk4(e.x * p[52] + p[56], e.y * p[53] + p[57],
              e.z * p[54] + p[58], e.w * p[55] + p[59]);
      emission = mk(emission.x * e.x, emission.y * e.y, emission.z * e.z);
    }
    if (matGraph && matGraph[mid * LIGHTUSD_GPU_GRAPH_FLOATS + 5] >= 0.f) {
      F4 g = evalMaterialXGraph(matGraph, mid, 4, u, v, ht, texels,
                                textures, numTextures);
      emission = mk(fmaxf(g.x, 0.f), fmaxf(g.y, 0.f), fmaxf(g.z, 0.f));
    }
    if (insts) {
      base = mk(base.x * insts[hitInst].tint[0],
                base.y * insts[hitInst].tint[1],
                base.z * insts[hitInst].tint[2]);
    }
  }
  return add(scale(base, 0.35f), emission);
}

__device__ F3 primaryLayerTransmissionTint(
    int tri, const int* mats, const float* matLightRt, int numMats) {
  if (!mats || !matLightRt || tri < 0) return mk(1.f, 1.f, 1.f);
  int mid = mats[tri];
  if (mid < 0 || mid >= numMats) return mk(1.f, 1.f, 1.f);
  const float* p = &matLightRt[mid * LIGHTUSD_GPU_OPENPBR_FLOATS];
  float w = fminf(fmaxf(p[11], 0.f), 1.f);
  float depth = fmaxf(p[15], 0.f);
  float dispersion=fminf(fmaxf(p[60]*p[62],0.f),1.f);
  F3 tint=mk(1.f+0.35f*dispersion,1.f,1.f-0.25f*dispersion);
  return mk(tint.x*expf(-w * depth * (fmaxf(0.f, 1.f - fminf(fmaxf(p[8], 0.f), 1.f)) + 0.25f * fmaxf(p[12], 0.f))),
            tint.y*expf(-w * depth * (fmaxf(0.f, 1.f - fminf(fmaxf(p[9], 0.f), 1.f)) + 0.25f * fmaxf(p[13], 0.f))),
            tint.z*expf(-w * depth * (fmaxf(0.f, 1.f - fminf(fmaxf(p[10], 0.f), 1.f)) + 0.25f * fmaxf(p[14], 0.f))));
}

__device__ F3 primaryLayerRefractedColor(
    int tri, float bu, float bv, int hitInst, const float* tris,
    const float* nrms,
    const Inst* insts, const int* mats, const float* matLightRt, int numMats,
    const Node* tlas, const Node* blas, const float* cols, const float* matBase,
    const int* backMats,
    const float* matPbr, const float* matGraph, const int* matTex,
  const float* matTexParam, const unsigned char* texels,
  const TexDesc* textures, int numTextures, const float* uvs,
  const float* uvs1, F3 rayOrigin, F3 rayDir, F3 missColor,
  F3* outOrigin, F3* outDir) {
  if (!tris || !insts || !mats || tri < 0 || hitInst < 0) return mk(0.f, 0.f, 0.f);
  if (!matLightRt) return mk(0.f, 0.f, 0.f);
  F3 throughput = mk(1.f, 1.f, 1.f);
  if (outOrigin) *outOrigin = rayOrigin;
  if (outDir) *outDir = rayDir;
  int currentTri = tri, currentInst = hitInst;
  float currentBu = bu, currentBv = bv;
  F3 currentDir = rayDir;
  // Match the Vulkan RT bounded walk: retain transmission through three
  // refractive interfaces before falling back to the approximate underlay.
  for (int bounce = 0; bounce < 3; ++bounce) {
    int mid = mats[currentTri];
    if (mid < 0 || mid >= numMats) return mk(0.f, 0.f, 0.f);
    F3 p0 = xfPt(insts[currentInst].o2w,
                 mk(tris[currentTri * 9 + 0], tris[currentTri * 9 + 1],
                    tris[currentTri * 9 + 2]));
    F3 p1 = xfPt(insts[currentInst].o2w,
                 mk(tris[currentTri * 9 + 3], tris[currentTri * 9 + 4],
                    tris[currentTri * 9 + 5]));
    F3 p2 = xfPt(insts[currentInst].o2w,
                 mk(tris[currentTri * 9 + 6], tris[currentTri * 9 + 7],
                    tris[currentTri * 9 + 8]));
    F3 geometricN = norm3(cross3(sub(p1, p0), sub(p2, p0)));
    F3 n = geometricN;
    if (nrms) {
      const float* nv = &nrms[currentTri * 9];
      float w0 = 1.f - currentBu - currentBv;
      F3 localN = mk(nv[0]*w0 + nv[3]*currentBu + nv[6]*currentBv,
                     nv[1]*w0 + nv[4]*currentBu + nv[7]*currentBv,
                     nv[2]*w0 + nv[5]*currentBu + nv[8]*currentBv);
      F3 smoothN = norm3(xfNrm(insts[currentInst].w2o, localN));
      if (dot3(smoothN, smoothN) > 1.0e-8f) {
        if (dot3(smoothN, geometricN) < 0.f) smoothN = scale(smoothN, -1.f);
        n = smoothN;
      }
    }
    const float normalDir = dot3(n, currentDir);
    const bool entering = normalDir < 0.f;
    if (normalDir > 0.f) n = scale(n, -1.f);
    const float ior = fmaxf(matLightRt[mid * LIGHTUSD_GPU_OPENPBR_FLOATS + 43], 1.f);
    const float eta = entering ? 1.f / ior : ior;
    const float cosi = -dot3(n, currentDir);
    const float k = 1.f - eta * eta * (1.f - cosi * cosi);
    F3 refr = k < 0.f ? sub(currentDir, scale(n, 2.f * dot3(currentDir, n)))
                      : add(scale(currentDir, eta),
                            scale(n, eta * cosi - sqrtf(k)));
    refr = norm3(refr);
    F3 tint = primaryLayerTransmissionTint(currentTri, mats, matLightRt,
                                           numMats);
    throughput = mk(throughput.x * tint.x, throughput.y * tint.y,
                     throughput.z * tint.z);
    F3 hit = add(add(scale(p0, 1.f - currentBu - currentBv),
                     scale(p1, currentBu)), scale(p2, currentBv));
    if (outOrigin) *outOrigin = add(hit, scale(refr, 1.0e-3f));
    if (outDir) *outDir = refr;
    int nextTri = -1, nextInst = -1; float nextBu = 0.f, nextBv = 0.f;
    traverseCutout(tlas, insts, blas, tris, add(hit, scale(refr, 1.0e-3f)),
                  refr, 1.0e30f, &nextTri, &nextBu, &nextBv, &nextInst,
                  cols, mats, backMats, matPbr, matLightRt, matGraph, numMats,
                  matTex, matTexParam, texels, textures, numTextures, uvs, uvs1,
                  -1, false);
    if (nextTri < 0 || nextInst < 0)
      return ptMul(throughput, missColor);
    currentTri = nextTri; currentInst = nextInst;
    currentBu = nextBu; currentBv = nextBv; currentDir = refr;
    int nextMid=mats[currentTri];
    float nextTransmission=0.f;
    if (nextMid>=0 && nextMid<numMats) {
      const float* nextP=&matLightRt[nextMid*LIGHTUSD_GPU_OPENPBR_FLOATS];
      nextTransmission=fminf(fmaxf(nextP[11],0.f),1.f);
      float nextOpacity=nextP[51]>0.5f?nextP[39]:matPbr[nextMid*6+5];
      if (nextP[54]>1.5f && nextP[43]>1.01f && nextOpacity<0.999f)
        nextTransmission=fmaxf(nextTransmission,1.f-nextOpacity);
    }
    if (nextTransmission<=1.0e-4f) {
      F3 result = approxPrimaryLayer(currentTri, currentBu, currentBv,
                                      currentInst, cols, mats, matBase,
                                      matPbr, matLightRt, matGraph, numMats, matTex,
                                      matTexParam, texels, textures,
                                      numTextures, uvs, insts);
      return mk(throughput.x * result.x, throughput.y * result.y,
                throughput.z * result.z);
    }
    if (bounce == 2) return ptMul(throughput, missColor);
  }
  return mk(0.f, 0.f, 0.f);
}

// Analytic Gaussian ellipse traversal. Point leaves use the same compact Node
// layout as the mesh BVH, but the primitive test stays native (no tessellation).
__device__ float pointHitBvh(const Node* bvh, const int* order,
                             const float* centers, const float* major,
                             const float* normals, const float* radii,
                             F3 o, F3 d, int count, int* hit,
                             float* gaussianWeight) {
  if (!bvh || !order || count <= 0) {
    *hit = -1; *gaussianWeight = 0.f; return 1e30f;
  }
  float inv[3] = {1.f/d.x, 1.f/d.y, 1.f/d.z};
  int stack[64], sp = 0; stack[sp++] = 0; float best = 1e30f; *hit = -1;
  *gaussianWeight = 0.f;
  while (sp > 0) {
    const Node* nd = &bvh[stack[--sp]];
    if (!hitAabb(nd, o, mk(inv[0], inv[1], inv[2]), best)) continue;
    if (nd->count > 0) {
      for (int j = 0; j < nd->count; ++j) {
        const int p = order[nd->left + j];
        F3 c = mk(centers[p*3], centers[p*3+1], centers[p*3+2]);
        F3 n = norm3(mk(normals[p*3], normals[p*3+1], normals[p*3+2]));
        F3 a = norm3(mk(major[p*3], major[p*3+1], major[p*3+2]));
        F3 b = norm3(cross3(n, a));
        float den = dot3(d, n);
        if (fabsf(den) < 1e-8f) continue;
        float t = dot3(sub(c, o), n) / den;
        if (t <= 0.f || t >= best) continue;
        F3 q = sub(add(o, scale(d, t)), c);
        float u = dot3(q, a) / radii[p*2];
        float v = dot3(q, b) / radii[p*2+1];
        const float r2 = u*u + v*v;
        if (r2 < 1.f) {
          best = t; *hit = p;
          // The carrier radii are 2 sigma. Keep the analytic support bounded
          // at 2 sigma while retaining the Gaussian density in the interior.
          *gaussianWeight = expf(-2.f * r2);
        }
      }
    } else {
      if (sp + 2 <= 64) { stack[sp++] = nd->left; stack[sp++] = nd->right; }
    }
  }
  return best;
}
__device__ float pointHitChunks(const Node* bvh, const int* order,
                                const PointBvhChunk* chunks, int chunkCount,
                                const float* centers, const float* major,
                                const float* normals, const float* radii,
                                F3 o, F3 d, int* hit, float* gaussianWeight) {
  float best = 1e30f;
  *hit = -1;
  *gaussianWeight = 0.f;
  if (!chunks || chunkCount <= 0) return best;
  for (int c = 0; c < chunkCount; ++c) {
    int localHit = -1;
    float localWeight = 0.f;
    const PointBvhChunk pc = chunks[c];
    const float t = pointHitBvh(
        bvh + pc.bvhFirst, order + pc.orderFirst, centers, major, normals,
        radii, o, d, pc.count, &localHit, &localWeight);
    if (localHit >= 0 && t < best) {
      best = t;
      *hit = localHit;
      *gaussianWeight = localWeight;
    }
  }
  return best;
}

  // Dome env background on miss (shaded mode only): sample the dome's latlong
  // envmap scaled by its effectiveColor. Packed light record: [0]=type,
  // [1]=flags (bit 4 = envmap mapped), [2]=texture id, [16..18]=effectiveColor.
  if (ht<0 && rmodePre==0 && lightParams && numLights>0 && texels && textures){
    for (int li=0; li<numLights; ++li){
      const float* lp=&lightParams[li*80];
      if ((int)(lp[0]+0.5f)!=6) continue;         // DomeLight
      int lflags=(int)(lp[1]+0.5f);
      if (!(lflags&(1<<4))) continue;             // has mapped envmap
      int envTex=(int)(lp[2]+0.5f);
      // Rows 10-12: world -> environment rotation (dome orientation).
      F3 de=mk(lp[40]*d.x+lp[41]*d.y+lp[42]*d.z,
               lp[44]*d.x+lp[45]*d.y+lp[46]*d.z,
               lp[48]*d.x+lp[49]*d.y+lp[50]*d.z);

      float ev=acosf(fminf(fmaxf(de.y,-1.f),1.f))*0.31830989f;
      F3 env=sampleDomeEnv(texels,textures,numTextures,envTex,eu,ev,
                           (lflags&(1<<8))!=0);
      outc=mk(env.x*lp[16],env.y*lp[17],env.z*lp[18]);
      break;
    }

  if (rmodePre==27 && pointHit < 0){  // BVH traversal-cost heatmap (per primary ray, all pixels)
    int steps=traverseSteps(tlas,insts,blas,tris,o,d,1e30f);
    float c=fminf(steps/256.0f,1.0f);  // 2-level visits more nodes than single BVH
    outc=mk(c, 1.f-fabsf(c-0.5f)*2.f, 1.f-c);
  } else if (ht>=0){
    const Inst* I=&insts[hitInst];
    float w0=1.f-bu-bv;
    F3 hit=add(o,scale(d,t));  // world hit (world ray + world t)
    // World-space triangle verts (geometric normal / facing / tangent / area AOVs).
    const float* tv=&tris[ht*9];

    F3 wp1=xfPt(I->o2w,mk(tv[3],tv[4],tv[5]));
    F3 wp2=xfPt(I->o2w,mk(tv[6],tv[7],tv[8]));

    float vv2=uvh[1]*w0+uvh[3]*bu+uvh[5]*bv;

      const float* uvh1=&uvs1[ht*6];

      vv21=uvh1[1]*w0+uvh1[3]*bu+uvh1[5]*bv;

    F3 sp1=projPix(cam.vp,wp1,(float)W,(float)H);
    F3 sp2=projPix(cam.vp,wp2,(float)W,(float)H);

                            (sp1.y-sp0.y)*(sp2.x-sp0.x));

      specIor=lrt[43];

                            (1.f-0.25f*fabsf(specAnisotropy)),

                                 transmissionDispersionScale,0.f),1.f);

            base.z*(1.f+(coatColor.z-1.f)*coatWeight*coatAffectColor));

              coatRoughness*fminf(fmaxf(coatWeight*coatAffectRoughness,0.f),1.f);
    float volumeOpacity=1.f-expf(-fmaxf(volumeDensity,0.f)*0.1f);

            base.y*(1.f+(volumeAlbedo.y-1.f)*volumeOpacity),
            base.z*(1.f+(volumeAlbedo.z-1.f)*volumeOpacity));
    emissionConst=add(emissionConst,scale(volumeEmission,
                                          fmaxf(volumeEmissionScale,0.f)*volumeOpacity));

    // and the loaders neutralize the matching constant to 1 whenever a texture
    // is bound, so constant*texel is unconditionally correct.
    int occlusionTex=-1, coatWeightTex=-1, coatColorTex=-1, coatRoughTex=-1;
    int specularColorTex=-1, coatNormalTex=-1;
    if (matTex && mid>=0 && mid<numMats){
      occlusionTex=matTex[mid*LIGHTUSD_GPU_MATERIAL_TEX_SLOTS+6]; coatWeightTex=matTex[mid*LIGHTUSD_GPU_MATERIAL_TEX_SLOTS+7];
      coatColorTex=matTex[mid*LIGHTUSD_GPU_MATERIAL_TEX_SLOTS+8]; coatRoughTex=matTex[mid*LIGHTUSD_GPU_MATERIAL_TEX_SLOTS+9];
      specularColorTex=matTex[mid*LIGHTUSD_GPU_MATERIAL_TEX_SLOTS+10];
      coatNormalTex=matTex[mid*LIGHTUSD_GPU_MATERIAL_TEX_SLOTS+11];
    }
    float occluu=uu,occlvv=vv2,cwuu=uu,cwvv=vv2;
    float ccuu=uu,ccvv=vv2,cruu=uu,crvv=vv2;
    if (tp){
      int uvbits2=(int)(tp[99]+0.5f);
      xformUv(tp,72,(uvbits2&1)?uu1:uu,(uvbits2&1)?vv21:vv2,&occluu,&occlvv);
      xformUv(tp,78,(uvbits2&2)?uu1:uu,(uvbits2&2)?vv21:vv2,&cwuu,&cwvv);
      xformUv(tp,84,(uvbits2&4)?uu1:uu,(uvbits2&4)?vv21:vv2,&ccuu,&ccvv);
      xformUv(tp,90,(uvbits2&8)?uu1:uu,(uvbits2&8)?vv21:vv2,&cruu,&crvv);
    }
    float fd=(fmaxf(specIor,1.f)-1.f)/(fmaxf(specIor,1.f)+1.f);
    F3 materialF0=mk(fd*fd+(base.x-fd*fd)*metallic,
                     fd*fd+(base.y-fd*fd)*metallic,
                     fd*fd+(base.z-fd*fd)*metallic);
    bool previewSpec=tp && tp[131]>0.5f;
    bool openPbrSpec=tp && tp[139]>0.5f;
    if (previewSpec || openPbrSpec) {
      F3 tint=lrt?mk(lrt[4],lrt[5],lrt[6]):mk(1.f,1.f,1.f);
      materialF0=previewSpec?tint:
          mk(fd*fd*tint.x+(base.x-fd*fd*tint.x)*metallic,
             fd*fd*tint.y+(base.y-fd*fd*tint.y)*metallic,
             fd*fd*tint.z+(base.z-fd*fd*tint.z)*metallic);
      float suu=uu,svv=vv2;
      xformUv(tp,124,tp[130]>0.5f?uu1:uu,tp[130]>0.5f?vv21:vv2,&suu,&svv);
      if (specularColorTex>=0) {
        F4 st=sampleMaterialTexture(texels,textures,numTextures,specularColorTex,
            suu,svv,textureLod(textures,numTextures,specularColorTex,suu,svv,
                               xformFootprint(tp,124,uvFootprint)), faces[ht],
            mk4(1.f,1.f,1.f,1.f));
        materialF0=mk(materialF0.x*(st.x*tp[132]+tp[136]),
                      materialF0.y*(st.y*tp[133]+tp[137]),
                      materialF0.z*(st.z*tp[134]+tp[138]));
      }
    }
    float nv=fmaxf(dot3(N,scale(d,-1.f)),1e-4f);
    materialF0=applyThinFilm(materialF0,nv,thinFilmWeight,thinFilmThickness,thinFilmIor);
    float occlusion=1.f;
    if (occlusionTex>=0){
      F4 at=sampleMaterialTexture(texels,textures,numTextures,occlusionTex,
          occluu,occlvv,
          textureLod(textures,numTextures,occlusionTex,occluu,occlvv,
                     xformFootprint(tp,72,uvFootprint)), faces[ht],
          mk4(1.f,1.f,1.f,1.f));
      int ch=tp?(int)(tp[96]+0.5f):0;
      occlusion=fminf(fmaxf(channel4(at,ch)*(tp?tp[70]:1.f)+(tp?tp[71]:0.f),

    }
    if (coatWeightTex>=0){
      F4 ct=sampleMaterialTexture(texels,textures,numTextures,coatWeightTex,

        F4 g=evalMaterialXGraphUv(matGraph,mid,4,euu,evv,uu1,vv21,faces[ht],texels,textures,numTextures,graphNormal,graphPosition,graphView,cam.context[0],cam.context[1]);

                         scale(gB,gn.y*2.f-1.f)),
                     scale(gN,gn.z*2.f-1.f)));
      }

    // Graph opacity is authoritative, but it must still obey the material's
    // authored alpha mode. Previously CUDA/HIP applied MASK before evaluating

    if (alphaMode > 0.5f && alphaMode < 1.5f)
      materialOpacity=(materialOpacity >= alphaCutoff) ? 1.f : 0.f;

    // a Blend atlas is opaque; only an authored fractional material remains
    // transmissive glass.
    if (alphaMode > 1.5f && opacityTex >= 0 && authoredOpacity >= 0.999f &&
        materialOpacity >= 0.5f)
      materialOpacity=1.f;
    if (alphaMode > 1.5f && specIor > 1.01f && authoredOpacity < 0.999f)
      // PreviewSurface represents implicit glass through fractional opacity
      // and IOR. Its transmission is the uncovered fraction, not a full
      // transmission override: the latter drops the authored shaded layer.
      transmission=fmaxf(transmission,1.f-materialOpacity);

    // packed material block while preserving the existing texture sampling and
    // debug-AOV behavior.

        }

                         lp[48]*Rb.x+lp[49]*Rb.y+lp[50]*Rb.z));
)CUDA"

// cap on its own; see the NOTE above.
R"CUDA(          float su=atan2f(Re.x,-Re.z)*0.15915494f+0.5f;
          float sv=acosf(fminf(fmaxf(Re.y,-1.f),1.f))*0.31830989f;

                               (((int)(lp[1]+0.5f))&(1<<8))!=0);

          specIBL3=mk(envc.x*(F0.x*dfgx+dfgy)*lp[16]*sw,
                      envc.y*(F0.y*dfgx+dfgy)*lp[17]*sw,
                      envc.z*(F0.z*dfgx+dfgy)*lp[18]*sw);
        }
        break;
      }
    }
    F3 em=mk(0.f,0.f,0.f);
    if (mid>=0&&mid<numMats) {
      em=emissionConst;
      F4 et4=sampleMaterialTexture(texels,textures,numTextures,emissiveTex,euu,evv,
          textureLod(textures,numTextures,emissiveTex,euu,evv,
                     xformFootprint(tp,24,uvFootprint)), faces[ht],
          mk4(1.f,1.f,1.f,1.f));
      if (tp) et4=mk4(et4.x*tp[52]+tp[56],et4.y*tp[53]+tp[57],et4.z*tp[54]+tp[58],et4.w*tp[55]+tp[59]);
      F3 et=mk(et4.x,et4.y,et4.z);
      em=mk(em.x*et.x,em.y*et.y,em.z*et.z);
    }
    F3 direct3=mk(0.f,0.f,0.f);
    bool hadDirect=false;
    nv=fmaxf(dot3(Ns,V),1e-4f);
    float directionalRoughness=roughness;
    for (int li=0; lightParams && li<numLights; ++li) {
      const float* lp=&lightParams[li*80];

      if (lt==6) continue;
      hadDirect=true;
      int lflags=(int)(lp[1]+0.5f);
      bool directLinked=li<32 ? ((insts[hitInst].directLightMask&(1u<<li))!=0)
                              : ((lflags&(1<<5))!=0);

                                 &lightPos,&emitterN,&emitter,&geometryArea)) continue;

        F3 toL=sub(lightPos,hit);

        }

        float soft=fminf(fmaxf(lp[25],0.f),1.f);
        float outer=cosf(angle),inner=cosf(angle*(1.f-soft));
        float den=fmaxf(inner-outer,1e-5f);
        float t=fminf(fmaxf((cc-outer)/den,0.f),1.f);
        shape=t*t*(3.f-2.f*t)*powf(fmaxf(cc,0.f),fmaxf(lp[26],0.f));
      }
      float nl=fmaxf(dot3(Ns,Ll),0.f); if (nl<=0.f || shape<=0.f) continue;
      if (lt==7) shape*=fmaxf(dot3(emitterN,scale(Ll,-1.f)),0.f);
      if (shape<=0.f) continue;

        int sht,shi; float sbu,sbv;
        F3 so=add(hit,scale(Ns,1e-3f));
        float st=traverseCutout(tlas,insts,blas,tris,so,Ll,
                    fmaxf(maxT-2e-3f,1e-3f),&sht,&sbu,&sbv,&shi,cols,mats,
                    backMats,matPbr,matLightRt,matGraph,numMats,matTex,matTexParam,
                    texels,textures,numTextures,uvs,uvs1,li,
                    (lflags&(1<<6))!=0);

      }
      F3 Hh=norm3(add(Ll,V)); float nh=fmaxf(dot3(Ns,Hh),0.f),vh=fmaxf(dot3(V,Hh),0.f);
      F3 f0=materialF0;
      F3 F=fresnelSchlick(vh,f0);
      float sg=ggxD(nh,roughness)*ggxG1(nv,roughness)*ggxG1(nl,roughness)/fmaxf(4.f*nv*nl,1e-5f)*fmaxf(specWeight,0.f);

      float diffuseWeight=(1.f-metallic)*(1.f-transmission)*(1.f-subsurface);
      F3 diffB=mk((1.f-F.x)*diffuseWeight*base.x*0.31830989f,
                  (1.f-F.y)*diffuseWeight*base.y*0.31830989f,
                  (1.f-F.z)*diffuseWeight*base.z*0.31830989f);
      float sheenScale=fminf(fmaxf(sheenWeight,0.f),1.f) *
          (0.5f+0.5f*fminf(fmaxf(sheenRoughness,0.f),1.f)) *
          powf(1.f-nl,5.f)*0.31830989f;
      F3 sheenB=scale(sheenColor,sheenScale);
      F3 tangent=fabsf(Ns.y)<0.9f?norm3(cross3(Ns,mk(0.f,1.f,0.f))):
                 norm3(cross3(Ns,mk(1.f,0.f,0.f)));
      F3 bitangent=norm3(cross3(Ns,tangent));
      float rot=specRotation*0.01745329252f, cs=cosf(rot), sn=sinf(rot);
      F3 rotatedT=add(scale(tangent,cs),scale(bitangent,sn));
      F3 rotatedB=add(scale(tangent,-sn),scale(bitangent,cs));
      float hT=dot3(Hh,rotatedT), hB=dot3(Hh,rotatedB);
      directionalRoughness=fminf(fmaxf(roughness *
          (1.f+specAnisotropy*(hT*hT-hB*hB)),0.02f),1.f);
      float radiusGain=fminf(fmaxf(sqrtf(fmaxf(dot3(subsurfaceRadius,
          mk(0.2126f,0.7152f,0.0722f))*subsurfaceScale,0.f))*8.f,0.f),1.f);
      float diffusionShape=(1.f-radiusGain)*(0.25f+0.75f*powf(1.f-nl,2.f))+
                           radiusGain*(0.25f+0.75f*(1.f-nl));
      F3 subsurfaceB=scale(subsurfaceColor,
                          fminf(fmaxf(subsurface,0.f),1.f) * diffusionShape *
                          0.31830989f);
      float cnl=fmaxf(dot3(coatNs,Ll),0.f),cnv=fmaxf(dot3(coatNs,V),1e-4f);
      float cnh=fmaxf(dot3(coatNs,Hh),0.f);

      float cg=ggxD(cnh,coatRoughness)*ggxG1(cnv,coatRoughness)*ggxG1(cnl,coatRoughness)/fmaxf(4.f*cnv*cnl,1e-5f)*fminf(fmaxf(coatWeight,0.f),1.f);

      F3 baseBrdf=add(add(add(scale(diffB,ds),scale(sheenB,ds)),
                              scale(subsurfaceB,ds)),
                       scale(specB,ss));

      F3 brdf=add(scale(baseBrdf,nl),scale(coatBrdf,cnl));
      F3 rad=scale(lt==7?emitter:mk(lp[12],lp[13],lp[14]),
                   attenuation*shape*shadow*iesFactor(lp,scale(Ll,-1.f))*
                   (lt==7?geometryArea:1.f));
      direct3=add(direct3,mk(brdf.x*rad.x,brdf.y*rad.y,brdf.z*rad.z));
    }
    if (!hadDirect) {
      F3 Ll=norm3(mk(cam.lightDir[0],cam.lightDir[1],cam.lightDir[2]));
      float nl=fmaxf(dot3(Ns,Ll),0.f); F3 Hh=norm3(add(Ll,V));
      float nh=fmaxf(dot3(Ns,Hh),0.f),vh=fmaxf(dot3(V,Hh),0.f);
      F3 f0=materialF0;
      F3 F=fresnelSchlick(vh,f0);
      float sg=ggxD(nh,directionalRoughness)*ggxG1(nv,directionalRoughness)*
               ggxG1(nl,directionalRoughness)/fmaxf(4.f*nv*nl,1e-5f)*
               fmaxf(specWeight,0.f);
      float sheenScale=fminf(fmaxf(sheenWeight,0.f),1.f) *
          (0.5f+0.5f*fminf(fmaxf(sheenRoughness,0.f),1.f)) *
          powf(1.f-nl,5.f)*0.31830989f;
      float dw=(1.f-metallic)*(1.f-transmission)*(1.f-subsurface);

                  sheenColor.x*sheenScale+F.x*sg)*nl,
                 ((1.f-F.y)*dw*base.y*0.31830989f+
                  sheenColor.y*sheenScale+F.y*sg)*nl,
                 ((1.f-F.z)*dw*base.z*0.31830989f+
                  sheenColor.z*sheenScale+F.z*sg)*nl);
    }
    // Occlusion modulates indirect/ambient only, never direct (matches raster).
    float ambientDiffuse=(1.f-metallic)*(1.f-transmission)*(1.f-subsurface);
    F3 ambientBase=mk(base.x*amb3.x*occlusion*ambientDiffuse,
                      base.y*amb3.y*occlusion*ambientDiffuse,
                      base.z*amb3.z*occlusion*ambientDiffuse);
    outc=add(add(add(add(ambientBase,direct3),scale(specIBL3,occlusion)),
                 scale(sheenColor,fminf(fmaxf(sheenWeight,0.f),1.f)*0.025f)),em);

    // 4=geometric normal, 5=uv set 0, 6=depth (t/depthScale in lightDir[3]).
    int rmode=(int)(cam.clear[3]+0.5f);
    if (rmode==0) {

        F3 topRefr=primaryLayerRefractedColor(
            ht,bu,bv,hitInst,tris,nrms,insts,mats,matLightRt,numMats,tlas,blas,
            cols,matBase,backMats,matPbr,matGraph,matTex,matTexParam,texels,
            textures,numTextures,uvs,uvs1,o,d,bg,&refractedOrigin,

                        throughColor.y*(1.f-topTransmission)+topRefr.y*topTransmission,
                        throughColor.z*(1.f-topTransmission)+topRefr.z*topTransmission);

        }
        for (int layer=0; layer<7 &&
             (throughput.x+throughput.y+throughput.z)>0.009f; ++layer) {

          F3 no=add(walkOrigin,scale(walkDir,cursor));
          float nt=traverseCutout(tlas,insts,blas,tris,no,walkDir,1.0e30f,
                                  &nht,&nbu,&nbv,&ninst,cols,mats,backMats,
                                  matPbr,matLightRt,matGraph,numMats,matTex,
                                  matTexParam,texels,textures,numTextures,
                                  uvs,uvs1,-1,false);

          int nmid=mats[nht];
          float na=1.f;
          float nam=0.f;
          if (nmid>=0 && nmid<numMats) {
            na=matPbr ? matPbr[nmid*6+5] : 1.f;
            if (matLightRt) {
              nam=matLightRt[nmid*LIGHTUSD_GPU_OPENPBR_FLOATS+54];
              if (matLightRt[nmid*LIGHTUSD_GPU_OPENPBR_FLOATS+51]>0.5f) na=matLightRt[nmid*LIGHTUSD_GPU_OPENPBR_FLOATS+39];
              if (nam>0.5f && nam<1.5f) na=1.f;
            }
            const float* nc=&cols[nht*12];
            float nw=1.f-nbu-nbv;
            na*=nc[3]*nw+nc[7]*nbu+nc[11]*nbv;
            na*=insts[ninst].tint[3];
          }
          na=fminf(fmaxf(na,0.f),1.f);
          F3 lc=approxPrimaryLayer(nht,nbu,nbv,ninst,cols,mats,matBase,
                                   matPbr,matLightRt,matGraph,numMats,matTex,
                                   matTexParam,texels,textures,numTextures,
                                   uvs,insts);
          accum=add(accum,mk(lc.x*throughput.x*na,
                             lc.y*throughput.y*na,
                             lc.z*throughput.z*na));
          F3 ltint=primaryLayerTransmissionTint(nht,mats,matLightRt,numMats);
          // The top-layer bounded walk already carries its changed direction
          // across subsequent interfaces. Underlayers here were discovered on
          // the original straight camera ray, so launching another refracted
          // walk from that original direction would describe a contradictory
          // path. Retain their bounded tint/alpha approximation only.
          throughput=mk(throughput.x*ltint.x*(1.f-na),
                        throughput.y*ltint.y*(1.f-na),
                        throughput.z*ltint.z*(1.f-na));
          walkOrigin=add(no,scale(walkDir,nt+1.0e-4f));
          cursor=1.0e-4f;
          if (nam>0.5f && nam<1.5f) break;
        }
        outc=add(accum,mk(throughColor.x*throughput.x,
                          throughColor.y*throughput.y,
                          throughColor.z*throughput.z));
      }
    }
    if (rmode==1){
      // Original-polygon wireframe with screen-space analytic AA (usdview look):
      // project the triangle's world verts to pixels and measure the sample's
      // pixel distance to each REAL polygon edge (emask drops triangulation
      // diagonals), then feather. emask==7 (no polygon data) draws all 3 edges.
      unsigned em = emask ? emask[ht] : 7u;
      if (em==0u) em=7u;
      float sampX=px+0.5f+cam.sceneMin[3], sampY=py+0.5f+cam.sceneExtent[3];
      F3 P0=projPix(cam.vp,wp0,(float)W,(float)H);
      F3 P1=projPix(cam.vp,wp1,(float)W,(float)H);
      F3 P2=projPix(cam.vp,wp2,(float)W,(float)H);
      float dmin=1e30f;
      if ((em&1u)&&P1.z>0.f&&P2.z>0.f) dmin=fminf(dmin,pixDistSeg(sampX,sampY,P1.x,P1.y,P2.x,P2.y));
      if ((em&2u)&&P2.z>0.f&&P0.z>0.f) dmin=fminf(dmin,pixDistSeg(sampX,sampY,P2.x,P2.y,P0.x,P0.y));
      if ((em&4u)&&P0.z>0.f&&P1.z>0.f) dmin=fminf(dmin,pixDistSeg(sampX,sampY,P0.x,P0.y,P1.x,P1.y));
      const float hw=1.3f;  // pixels: ~1px line + 1px AA edge (matches the GL path)
      float cov=1.f-fminf(fmaxf((dmin-(hw-1.f))/1.f,0.f),1.f);  // smooth-ish feather
      cov=cov*cov*(3.f-2.f*cov);  // smoothstep
      // Diagnostic wire color: never tint it with the scene/preview light.
      F3 wireCol=mk(0.04f,0.24f,0.10f);
      F3 bg=mk(cam.clear[0],cam.clear[1],cam.clear[2]);
      outc=mk(bg.x+(wireCol.x-bg.x)*cov, bg.y+(wireCol.y-bg.y)*cov, bg.z+(wireCol.z-bg.z)*cov);
    } else if (rmode==2){
      outc=mk(N.x*0.5f+0.5f, N.y*0.5f+0.5f, N.z*0.5f+0.5f);
    } else if (rmode==35){
      outc=mk(coatN.x*0.5f+0.5f, coatN.y*0.5f+0.5f, coatN.z*0.5f+0.5f);
    } else if (rmode==36){
      float v=fminf(fmaxf(coatWeight,0.f),1.f); outc=mk(v,v,v);
    } else if (rmode==37){
      outc=mk(fminf(fmaxf(coatColor.x,0.f),1.f),
              fminf(fmaxf(coatColor.y,0.f),1.f),
              fminf(fmaxf(coatColor.z,0.f),1.f));
    } else if (rmode==38){
      float v=fminf(fmaxf(coatRoughness,0.f),1.f); outc=mk(v,v,v);
    } else if (rmode==39){
      outc=mk(fminf(fmaxf(materialF0.x,0.f),1.f),
              fminf(fmaxf(materialF0.y,0.f),1.f),
              fminf(fmaxf(materialF0.z,0.f),1.f));
    } else if (rmode==40){
      float ior=fmaxf(specIor,1.f),d=(ior-1.f)/(ior+1.f);
      outc=mk(d*d,d*d,d*d);
    } else if (rmode==3){
      outc=idColor(mid);
    } else if (rmode==4){
      F3 Ng=norm3(cross3(sub(wp1,wp0),sub(wp2,wp0)));
      if (dot3(Ng,d)>0.f) Ng=scale(Ng,-1.f);
      outc=mk(Ng.x*0.5f+0.5f, Ng.y*0.5f+0.5f, Ng.z*0.5f+0.5f);
    } else if (rmode==5){
      outc=mk(uu-floorf(uu), vv2-floorf(vv2), 0.f);
    } else if (rmode==6){
      float dd=fminf(fmaxf(t/fmaxf(cam.lightDir[3],1e-3f),0.f),1.f);
      outc=mk(1.f-dd,1.f-dd,1.f-dd);
    } else if (rmode==7){
      outc=base;  // albedo (pre-lighting)
    } else if (rmode==8){  // facing
      outc=(dot3(cross3(sub(wp1,wp0),sub(wp2,wp0)),d)<0.f)?mk(0.1f,0.7f,0.1f):mk(0.7f,0.1f,0.1f);
    } else if (rmode==9){  // roughness
      outc=mk(roughness,roughness,roughness);
    } else if (rmode==10){  // metallic
      outc=mk(metallic,metallic,metallic);
    } else if (rmode==11){  // emissive
      outc=em;
    } else if (rmode==12){  // opacity
      outc=mk(materialOpacity,materialOpacity,materialOpacity);
    } else if (rmode==13){  // world position
      outc=mk(fminf(fmaxf((hit.x-cam.sceneMin[0])/fmaxf(cam.sceneExtent[0],1e-4f),0.f),1.f),
              fminf(fmaxf((hit.y-cam.sceneMin[1])/fmaxf(cam.sceneExtent[1],1e-4f),0.f),1.f),
              fminf(fmaxf((hit.z-cam.sceneMin[2])/fmaxf(cam.sceneExtent[2],1e-4f),0.f),1.f));
    } else if (rmode==23){  // uv checker
      float cx=floorf((uu-floorf(uu))*16.f), cy=floorf((vv2-floorf(vv2))*16.f);
      float kk=fmodf(cx+cy,2.f); float g=0.25f+0.6f*kk; outc=mk(g,g,g);
    } else if (rmode==22){  // tangent (from triangle UV gradient)
      const float* uv=&uvs[ht*6];
      F3 e1=sub(wp1,wp0), e2=sub(wp2,wp0);
      float d1x=uv[2]-uv[0], d1y=uv[3]-uv[1], d2x=uv[4]-uv[0], d2y=uv[5]-uv[1];
      float r=d1x*d2y-d2x*d1y;
      F3 T = (fabsf(r)>1e-8f) ? scale(sub(scale(e1,d2y),scale(e2,d1y)),1.f/r) : e1;
      T=norm3(T); outc=mk(T.x*0.5f+0.5f,T.y*0.5f+0.5f,T.z*0.5f+0.5f);
    } else if (rmode==14){  // barycentric
      outc=mk(w0,bu,bv);
    } else if (rmode==15){  // prim id
      outc=idColor(ht);
    } else if (rmode==26){  // instance id (TLAS instance) -- matches VK-RT
      outc=idColor(I->instId);
    } else if (rmode==19){  // missing normals
      outc=(geo[ht]&1)?mk(0.95f,0.1f,0.85f):mk(0.2f,0.2f,0.2f);
    } else if (rmode==18){  // purpose (bits1-2 of geo byte)
      int p=(geo[ht]>>1)&3;
      outc=(p==1)?mk(0.2f,0.8f,0.3f):(p==2)?mk(0.2f,0.45f,0.95f):(p==3)?mk(0.95f,0.75f,0.1f):mk(0.5f,0.5f,0.5f);
    } else if (rmode==29){  // kind (bits3-5 of geo byte)
      int k=(geo[ht]>>3)&7;
      outc=(k==1)?mk(0.2f,0.8f,0.8f):(k==2)?mk(0.85f,0.3f,0.85f):(k==3)?mk(0.95f,0.6f,0.15f):(k==4)?mk(0.5f,0.85f,0.4f):mk(0.35f,0.35f,0.35f);
    } else if (rmode==30){  // udim tile from UV set 0
      int tile=int(floorf(uu))+10*int(floorf(vv2));
      outc=idColor(tile);
    } else if (rmode==33){  // texel density (per-triangle UV/world area ratio)
      const float* uv=&uvs[ht*6];
      F3 cr=cross3(sub(wp1,wp0),sub(wp2,wp0));
      float worldArea=sqrtf(dot3(cr,cr));
      float d1x=uv[2]-uv[0], d1y=uv[3]-uv[1], d2x=uv[4]-uv[0], d2y=uv[5]-uv[1];
      float uvArea=fabsf(d1x*d2y-d2x*d1y);
      float td=sqrtf(uvArea/fmaxf(worldArea,1e-12f));
      float c=fminf(fmaxf(td*cam.lightDir[3]*0.5f,0.f),1.f);
      outc=mk(c, 1.f-fabsf(c-0.5f)*2.f, 1.f-c);
    } else if (rmode==34){  // source USD face id
      outc=idColor(faces[ht]);
    } else if (rmode==31){  // uv set 1 (multi-UV)
      const float* uv=&uvs1[ht*6];
      float uu=uv[0]*w0+uv[2]*bu+uv[4]*bv, vv2=uv[1]*w0+uv[3]*bu+uv[5]*bv;
      outc=mk(uu-floorf(uu), vv2-floorf(vv2), 0.f);
    } else if (rmode==32){  // blendshape influence
      const float* f=&infls[ht*3];
      float infl=f[0]*w0+f[1]*bu+f[2]*bv;
      float c=fminf(fmaxf(infl/fmaxf(cam.lightDir[3]*0.1f,1e-4f),0.f),1.f);
      outc=mk(c, 1.f-fabsf(c-0.5f)*2.f, 1.f-c);
    } else if (rmode==21){  // skin weights: dominant joint tinted by its weight
      int dj=domj[ht];                          // provoking-vertex dominant joint (flat)
      const float* dw=&domw[ht*3];
      float wt=dw[0]*w0+dw[1]*bu+dw[2]*bv;        // interpolated dominant weight
      F3 ic=idColor(dj);
      float k2=0.3f+0.7f*fminf(fmaxf(wt,0.f),1.f);
      outc=(dj<0)?mk(0.45f,0.45f,0.45f):mk(ic.x*k2,ic.y*k2,ic.z*k2);
    } else if (rmode==25){  // curvature (geometric: vertex-normal spread across the tri)
      if (geo[ht]&1){
        outc=mk(0.f,1.f,1.f);                     // flat-shaded mesh -> zero curvature
      } else {
        const float* nv=&nrms[ht*9];
        F3 n0=norm3(mk(nv[0],nv[1],nv[2])), n1=norm3(mk(nv[3],nv[4],nv[5])), n2=norm3(mk(nv[6],nv[7],nv[8]));
        float spread=(1.f-dot3(n0,n1))+(1.f-dot3(n1,n2))+(1.f-dot3(n2,n0));
        float c=fminf(fmaxf(spread*4.f,0.f),1.f);
        outc=mk(c, 1.f-fabsf(c-0.5f)*2.f, 1.f-c);
      }
    } else if (rmode==24){  // ray-traced ambient occlusion
      F3 Ng=norm3(cross3(sub(wp1,wp0),sub(wp2,wp0)));
      if (dot3(Ng,d)>0.f) Ng=scale(Ng,-1.f);
      float aoR=fmaxf(cam.lightDir[3],1e-3f)*0.15f;
      unsigned seed=hashU((unsigned)px*1973u+(unsigned)py*9277u+1u);
      const int NS=24; float occ=0.f;
      for (int s=0;s<NS;s++){
        F3 sd=cosHemi(Ng,rndf(seed),rndf(seed));
        F3 so=add(hit,scale(Ng,aoR*1e-2f));
        int sht,si; float su,sv;
        traverseCutout(tlas,insts,blas,tris,so,sd,aoR,&sht,&su,&sv,&si,
                       cols,mats,backMats,matPbr,matLightRt,matGraph,numMats,matTex,
                       matTexParam,
                       texels,textures,numTextures,uvs,uvs1,-1,false);
        if (sht>=0) occ+=1.f;
      }
      float a=1.f-occ/float(NS); outc=mk(a,a,a);
    } else if (rmode==28){  // soft shadow / sky visibility toward the light
      F3 Lc=norm3(mk(cam.lightDir[0],cam.lightDir[1],cam.lightDir[2]));
      F3 tt=norm3(fabsf(Lc.x)>0.9f?mk(0.f,1.f,0.f):mk(1.f,0.f,0.f));
      F3 bb=norm3(cross3(Lc,tt)); tt=cross3(bb,Lc);
      unsigned seed=hashU((unsigned)px*2719u+(unsigned)py*5051u+7u);
      const int NS=16; float vis=0.f; const float coneR=0.08f;
      for (int s=0;s<NS;s++){
        float ang=rndf(seed)*6.2831853f, rr=sqrtf(rndf(seed))*coneR;
        F3 sd=norm3(add(Lc,add(scale(tt,rr*cosf(ang)),scale(bb,rr*sinf(ang)))));
        F3 so=add(hit,scale(N,1e-3f));
        int sht,si; float su,sv;
        traverseCutout(tlas,insts,blas,tris,so,sd,1e30f,&sht,&su,&sv,&si,
                       cols,mats,backMats,matPbr,matLightRt,matGraph,numMats,matTex,
                       matTexParam,
                       texels,textures,numTextures,uvs,uvs1,-1,false);
        if (sht<0) vis+=1.f;
      }
      float v=vis/float(NS); outc=mk(v,v,v);
    }
  }
  if (pointHit >= 0) {
    F3 N = norm3(mk(pointNormals[pointHit*3], pointNormals[pointHit*3+1],
                    pointNormals[pointHit*3+2]));
    if (dot3(N, d) > 0.f) N = scale(N, -1.f);
    F3 C = mk(pointColors[pointHit*4], pointColors[pointHit*4+1],
              pointColors[pointHit*4+2]);
    float alpha = fminf(fmaxf(pointColors[pointHit*4+3] * pointWeight, 0.f), 1.f);
    if (rmodePre == 2 || rmodePre == 4) outc = scale(add(N, mk(1.f,1.f,1.f)), 0.5f);
    else if (rmodePre == 6) outc = mk(fminf(t / fmaxf(cam.lightDir[3], 1e-6f), 1.f), 0.f, 0.f);
    else if (rmodePre == 0) {
      float lam = fmaxf(dot3(N, norm3(mk(cam.lightDir[0],cam.lightDir[1],cam.lightDir[2]))), 0.f);
      outc = add(scale(C, 0.18f + 0.82f * lam), scale(mk(cam.clear[0],cam.clear[1],cam.clear[2]), 1.f-alpha));
    } else outc = C;
  }
  // UsdVol volumes: composite over the shaded color (default mode only).
  if (rmodePre==0){
    for (int vi=0; vi<numVols; vi++){
      outc=compositeVolume(volDens, &volParams[vi], o, d, outc, &t);
    }
  }
  if (rmodePre==0) {
    float exposureScale=exp2f(cam.camPos[3]);
    outc=scale(outc,exposureScale);
  }
  if (rmodePre==0 || rmodePre==1) {
    if (rmodePre==0) outc=mk(aces1(outc.x),aces1(outc.y),aces1(outc.z));
    outc=mk(linearToSrgb1(outc.x),linearToSrgb1(outc.y),linearToSrgb1(outc.z));
  }
  int idx=(py*W+px)*4;
  if (hitDepth) hitDepth[py*W+px] = t;
  unsigned char qr=(unsigned char)(fminf(fmaxf(outc.x,0.f),1.f)*255.f+0.5f);
  unsigned char qg=(unsigned char)(fminf(fmaxf(outc.y,0.f),1.f)*255.f+0.5f);
  unsigned char qb=(unsigned char)(fminf(fmaxf(outc.z,0.f),1.f)*255.f+0.5f);
  if (numSamples<=1 || !accum){
    out[idx+0]=qr; out[idx+1]=qg; out[idx+2]=qb; out[idx+3]=255;
    return;
  }
  // Supersampled AA: accumulate the QUANTIZED sample on the device (identical
  // math to the old host loop, so the averaged image is byte-identical) and
  // resolve on the last sample -- one readback for the whole frame instead of
  // one per sample.
  float ar=(sampleIdx==0)?(float)qr:accum[idx+0]+(float)qr;
  float ag=(sampleIdx==0)?(float)qg:accum[idx+1]+(float)qg;
  float ab=(sampleIdx==0)?(float)qb:accum[idx+2]+(float)qb;
  accum[idx+0]=ar; accum[idx+1]=ag; accum[idx+2]=ab;
  if (sampleIdx==numSamples-1){
    out[idx+0]=(unsigned char)(ar/(float)numSamples+0.5f);
    out[idx+1]=(unsigned char)(ag/(float)numSamples+0.5f);
    out[idx+2]=(unsigned char)(ab/(float)numSamples+0.5f);
    out[idx+3]=255;
  }
}
